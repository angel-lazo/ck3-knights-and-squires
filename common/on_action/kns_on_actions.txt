on_game_start = {
	on_actions = {
		kns_on_game_start
	}
}

kns_on_game_start = {
	effect = {
		every_ruler = {
			limit = {
				is_landed = yes
				kns_practices_knighthood_trigger = yes
			}
			kns_try_make_starting_knight_effect = yes
			every_courtier = {
				limit = {
					is_adult = yes
					kns_practices_knighthood_trigger = yes
				}
				kns_try_make_starting_knight_effect = yes
			}
			every_courtier = {
				limit = {
					kns_is_eligible_for_squirehood_trigger = yes
					OR = {
						is_close_family_of = prev
						is_child_of = prev
					}
				}
				kns_try_assign_starting_squire_effect = yes
			}
		}
		kns_recount_living_prodigies_effect = yes
		kns_try_start_papal_youth_tourney_effect = yes
	}
}

on_death = {
	on_actions = {
		kns_on_death
	}
}

kns_on_death = {
	effect = {
		if = {
			limit = {
				OR = {
					has_character_flag = kns_is_imperial_guard
					has_character_modifier = kns_imperial_guard_modifier
				}
			}
			if = {
				limit = { is_acclaimed = yes }
				save_scope_as = kns_ig_dying
				accolade = {
					kns_ig_stamp_accolade_from_knight_effect = { KNIGHT = scope:kns_ig_dying }
				}
			}
			kns_dismiss_imperial_guard_effect = { KNIGHT = this }
		}
		if = {
			limit = { num_of_relation_kns_imperial_guard > 0 }
			kns_dismiss_all_imperial_guards_effect = yes
		}
		if = {
			limit = { exists = var:kns_funding_patron }
			var:kns_funding_patron = { save_scope_as = kns_dead_patron }
			remove_character_flag = kns_prowess_funded
			remove_variable = kns_funding_patron
			scope:kns_dead_patron ?= {
				kns_refresh_training_upkeep_effect = yes
			}
		}
		if = {
			limit = { has_trait = kns_talent_generation }
			kns_unregister_living_prodigy_effect = yes
			if = {
				limit = {
					NOT = { has_character_flag = kns_papal_tourney_prodigy }
					NOT = { kns_living_prodigious_exists_trigger = yes }
				}
				save_scope_as = kns_dead_prodigious
				kns_try_start_papal_youth_tourney_effect = yes
			}
		}
		if = {
			limit = { exists = var:kns_training_knight_count }
			every_courtier = {
				limit = { has_character_flag = kns_prowess_funded }
				remove_character_flag = kns_prowess_funded
				remove_variable = kns_funding_patron
				kns_update_training_modifier_effect = yes
			}
			every_vassal = {
				limit = { has_character_flag = kns_prowess_funded }
				remove_character_flag = kns_prowess_funded
				remove_variable = kns_funding_patron
				kns_update_training_modifier_effect = yes
			}
		}
		if = {
			limit = {
				OR = {
					AND = {
						has_variable = kns_imperial_duel_wins
						var:kns_imperial_duel_wins > 0
					}
					AND = {
						has_variable = kns_imperial_titles_won
						var:kns_imperial_titles_won > 0
					}
				}
			}
			# on_death often still has is_alive = yes. Execution also prunes
			# unimportant dead, which drops them from kns_hof_index.
			add_character_flag = kns_hof_freezing
			make_unprunable = yes
			kns_register_imperial_hof_character_effect = { CHAR = this }
			# on_death is "about to die". Freeze again tomorrow when is_alive is no.
			trigger_event = { id = kns_hof.0001 days = 1 }
			every_player = {
				trigger_event = { id = kns_hof.0001 days = 1 }
			}
		}
		if = {
			limit = {
				OR = {
					has_character_flag = kns_olympic_institution
					has_character_flag = kns_imperial_olympic_host
				}
			}
			kns_handle_imperial_olympic_organizer_death_effect = yes
		}
	}
}

	on_actions = {

	}

kns_quarterly_playable_pulse = {
	effect = {
		if = {
			limit = {
				OR = {
					num_of_relation_kns_imperial_guard > 0
					exists = var:kns_imperial_guard_window_open
				}
			}
			kns_sweep_invalid_imperial_guards_effect = yes
		}
		if = {
			limit = {
				OR = {
					exists = var:kns_training_knight_count
					has_character_modifier = kns_funding_knight_training
					has_character_flag = kns_prowess_funded
				}
			}
			kns_sweep_patron_old_age_funding_effect = yes
			kns_update_training_modifier_effect = yes
			every_courtier = {
				limit = {
					OR = {
						has_character_flag = kns_prowess_funded
						has_character_modifier = kns_in_prowess_training
					}
				}
				kns_update_training_modifier_effect = yes
			}
			every_vassal = {
				limit = {
					OR = {
						has_character_flag = kns_prowess_funded
						has_character_modifier = kns_in_prowess_training
					}
				}
				kns_update_training_modifier_effect = yes
			}
			kns_refresh_training_upkeep_effect = yes
		}
	}
	random_events = {
		850 = 0
		75 = kns_knighthood.0016
		75 = kns_knighthood.0017
	}
}
