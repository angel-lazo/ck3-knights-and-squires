# Finals round on the host (128-bracket round 7, 256-bracket round 8).
kns_olympic_host_is_finals_round_trigger = {
	has_variable = kns_olympic_round
	OR = {
		AND = {
			has_variable = kns_olympic_bracket_size
			var:kns_olympic_bracket_size = 256
			var:kns_olympic_round = 8
		}
		AND = {
			has_variable = kns_olympic_bracket_size
			var:kns_olympic_bracket_size = 128
			var:kns_olympic_round = 7
		}
		var:kns_olympic_round >= 8
	}
}
