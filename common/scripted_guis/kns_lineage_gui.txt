kns_lineage_window_is_shown = {
	scope = character
	is_shown = {
		exists = var:kns_lineage_subject
	}
}

kns_lineage_has_teacher = {
	scope = character
	is_shown = {
		exists = var:kns_lineage_teacher
		var:kns_lineage_subject ?= {
			NOT = { has_character_flag = kns_bought_knighthood }
		}
	}
}

kns_lineage_is_purchased = {
	scope = character
	is_shown = {
		var:kns_lineage_subject ?= {
			has_character_flag = kns_bought_knighthood
		}
	}
}
