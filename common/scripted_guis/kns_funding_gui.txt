kns_gui_toggle_knight_funding = {
	scope = character
	saved_scopes = {
		target
	}
	is_valid = {
		is_imprisoned = no
		OR = {
			scope:target = {
				kns_is_funded_by_character_trigger = { PATRON = root }
			}
			scope:target = {
				kns_can_be_funded_by_trigger = { PATRON = root }
			}
		}
	}
	effect = {
		custom_tooltip = kns_funding_window_toggle_tt
		if = {
			limit = {
				scope:target = {
					kns_is_funded_by_character_trigger = { PATRON = root }
				}
			}
			kns_stop_funding_knight_effect = {
				KNIGHT = scope:target
				PATRON = root
			}
		}
		else_if = {
			limit = {
				scope:target = {
					kns_can_be_funded_by_trigger = { PATRON = root }
				}
			}
			kns_start_funding_knight_effect = {
				KNIGHT = scope:target
				PATRON = root
			}
		}
		kns_rebuild_funding_knight_list_effect = yes
	}
}
