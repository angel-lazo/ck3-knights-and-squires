kns_gui_ig_can_appoint = {
	scope = character
	saved_scopes = {
		target
	}
	is_valid = {
		is_imprisoned = no
		gold >= 1500
		kns_emperor_has_imperial_guard_slot_trigger = yes
		scope:target = {
			kns_can_be_appointed_imperial_guard_by_trigger = { EMPEROR = root }
		}
	}
	effect = {
		custom_tooltip = kns_imperial_guard_kit_tt
		remove_short_term_gold = 1500
		if = {
			limit = {
				has_variable = kns_ig_hiring_axe
				NOT = { exists = var:kns_ig_slot_axe }
			}
			kns_appoint_imperial_guard_effect = {
				KNIGHT = scope:target
				EMPEROR = root
				WEAPON = axe
			}
		}
		else_if = {
			limit = {
				has_variable = kns_ig_hiring_hammer
				NOT = { exists = var:kns_ig_slot_hammer }
			}
			kns_appoint_imperial_guard_effect = {
				KNIGHT = scope:target
				EMPEROR = root
				WEAPON = hammer
			}
		}
		else_if = {
			limit = {
				has_variable = kns_ig_hiring_sword
				NOT = { exists = var:kns_ig_slot_sword }
			}
			kns_appoint_imperial_guard_effect = {
				KNIGHT = scope:target
				EMPEROR = root
				WEAPON = sword
			}
		}
		else_if = {
			limit = {
				has_variable = kns_ig_hiring_spear
				NOT = { exists = var:kns_ig_slot_spear }
			}
			kns_appoint_imperial_guard_effect = {
				KNIGHT = scope:target
				EMPEROR = root
				WEAPON = spear
			}
		}
		else_if = {
			limit = {
				has_variable = kns_ig_hiring_mace
				NOT = { exists = var:kns_ig_slot_mace }
			}
			kns_appoint_imperial_guard_effect = {
				KNIGHT = scope:target
				EMPEROR = root
				WEAPON = mace
			}
		}
		else = {
			kns_appoint_imperial_guard_to_first_vacant_effect = { KNIGHT = scope:target }
		}
		set_variable = {
			name = kns_ig_attire_target
			value = scope:target
		}
		send_interface_toast = {
			type = event_toast_effect_good
			title = kns_appoint_imperial_guard_toast
			custom_tooltip = kns_appoint_imperial_guard_toast.tt
			left_icon = scope:target
		}
		kns_close_imperial_guard_hire_effect = yes
		kns_rebuild_imperial_guard_window_effect = yes
	}
}
