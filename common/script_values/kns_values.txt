# Fake-duel random_list modifier, evaluated on the fighter.
# Total weight = 5 (list base) + max(0, prowess×2 − 5×floor(gap/10)).
# Gap penalty applies only when this fighter is behind the other.
# Outer min 0 keeps the add from going negative so the +5 floor is always intact.
kns_olympic_fake_duel_prowess_weight_value = {
	value = prowess
	multiply = 2
	if = {
		limit = {
			exists = scope:kns_olympic_fighter_a
			exists = scope:kns_olympic_fighter_b
			this = scope:kns_olympic_fighter_a
			prowess < scope:kns_olympic_fighter_b.prowess
		}
		subtract = {
			value = scope:kns_olympic_fighter_b.prowess
			subtract = prowess
			min = 0
			divide = 10
			floor = yes
			multiply = 5
		}
	}
	if = {
		limit = {
			exists = scope:kns_olympic_fighter_a
			exists = scope:kns_olympic_fighter_b
			this = scope:kns_olympic_fighter_b
			prowess < scope:kns_olympic_fighter_a.prowess
		}
		subtract = {
			value = scope:kns_olympic_fighter_a.prowess
			subtract = prowess
			min = 0
			divide = 10
			floor = yes
			multiply = 5
		}
	}
	min = 0
}

# Fake-duel list total: 5+weight per fighter.
kns_olympic_fake_total_weight_value = {
	value = 10
	if = {
		limit = { exists = scope:kns_olympic_fighter_a }
		add = scope:kns_olympic_fighter_a.kns_olympic_fake_duel_prowess_weight_value
	}
	if = {
		limit = { exists = scope:kns_olympic_fighter_b }
		add = scope:kns_olympic_fighter_b.kns_olympic_fake_duel_prowess_weight_value
	}
	min = 1
}

# Percent chance this character wins the fake duel. Evaluate on the combatant.
kns_olympic_fake_player_win_percent_value = {
	value = 5
	add = kns_olympic_fake_duel_prowess_weight_value
	multiply = 100
	divide = kns_olympic_fake_total_weight_value
	round = yes
	min = 0
	max = 100
}
