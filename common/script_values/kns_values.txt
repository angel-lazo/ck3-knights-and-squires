# 50 gold per worn weapon or armor piece at or below 15% durability.
kns_olympic_gear_repair_gold_value = {
	value = 0
	if = {
		limit = {
			any_equipped_character_artifact = {
				artifact_slot_type = primary_armament
				artifact_durability_percent <= 0.15
			}
		}
		add = 50
	}
	if = {
		limit = {
			any_equipped_character_artifact = {
				artifact_slot_type = armor
				artifact_durability_percent <= 0.15
			}
		}
		add = 50
	}
}
