kns_offer_knighthood_interaction = {
	category = interaction_category_diplomacy
	desc = kns_offer_knighthood_interaction_desc
	icon = icon_combat

	ai_maybe = yes
	ai_min_reply_days = 1
	ai_max_reply_days = 3
	popup_on_receive = yes

	greeting = positive
	send_name = kns_offer_of_knighthood
	notification_text = KNS_OFFER_KNIGHTHOOD_NOTIFICATION

	use_diplomatic_range = yes

	cost = {
		prestige = { value = medium_prestige_value }
	}

	is_shown = {
		scope:actor = {
			kns_actor_can_offer_knighthood_trigger = yes
			NOT = { this = scope:recipient }
		}
		scope:recipient = {
			kns_can_receive_knighthood_offer_from_trigger = { ACTOR = scope:actor }
		}
	}

	on_accept = {
		kns_become_a_knight_effect = {
			KNIGHT_MAKER = scope:actor
			KNIGHT_TO_BE = scope:recipient
		}
	}

	on_decline = {
		scope:actor = {
			add_prestige = miniscule_prestige_loss
			add_opinion = {
				target = scope:recipient
				modifier = kns_declined_knighthood_opinion
				opinion = -15
			}
			send_interface_toast = {
				title = kns_knighthood_offer_rejected_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				custom_tooltip = kns_knighthood_offer_rejected_toast.tt
			}
		}
	}

	ai_frequency = 24

	ai_potential = {
		is_adult = yes
		kns_is_knight_trigger = yes
		is_available_ai_adult = yes
	}

	ai_targets = {
		ai_recipients = children
		ai_recipients = courtiers
		ai_recipients = vassals
		max = 5
	}

	ai_targets = {
		ai_recipients = scripted_relations
	}

	auto_accept = {
		custom_description = {
			text = "KNS_THEY_ARE_MY_SQUIRE"
			subject = scope:actor
			object = scope:recipient
			scope:recipient = {
				has_relation_kns_knight = scope:actor
			}
		}
	}

	ai_accept = {
		base = 20
		modifier = {
			add = { value = ai_boldness divide = 2 min = 10 }
			ai_boldness > 0
			desc = KNS_KNIGHTHOOD_HIGH_BOLDNESS_REASON
		}
		modifier = {
			add = 10
			has_education_martial_trigger = yes
			desc = KNS_KNIGHTHOOD_EDUCATION_MARTIAL_REASON
		}
		modifier = {
			add = -15
			has_education_learning_trigger = yes
			desc = KNS_KNIGHTHOOD_EDUCATION_LEARNING_REASON
		}
		modifier = {
			add = 25
			has_trait_rank = {
				trait = education_martial_prowess
				rank >= 1
			}
			desc = KNS_KNIGHTHOOD_I_AM_A_WARRIOR_REASON
		}
		modifier = {
			add = -200
			NOT = { kns_practices_knighthood_trigger = yes }
			desc = KNS_KNIGHTHOOD_MY_FAITH_DOES_NOT_CARE
		}
		modifier = {
			add = 20
			kns_practices_knighthood_trigger = yes
			desc = KNS_KNIGHTHOOD_HAS_KNIGHTHOOD_REASON
		}
		modifier = {
			add = -1000
			OR = {
				has_relation_rival = scope:actor
				has_relation_nemesis = scope:actor
			}
			desc = AI_YOUR_RIVAL
		}
		modifier = {
			add = 50
			is_lowborn = yes
			scope:actor = { is_lowborn = no }
			desc = KNS_LOWBORN_DESIRES_KNIGHTHOOD_REASON
		}
		modifier = {
			add = -100
			is_lowborn = no
			scope:actor = { is_lowborn = yes }
			desc = KNS_HIGHBORN_DISLIKES_LOWBORN_KNIGHT_OFFER
		}
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.25
			desc = AI_SIMPLE_OPINION_REASON
		}
	}

	ai_will_do = {
		base = 0
		modifier = {
			add = 2000
			has_relation_kns_squire = scope:recipient
		}
		modifier = {
			add = 20
			kns_practices_knighthood_trigger = yes
		}
		modifier = {
			add = 5
			OR = {
				has_trait = gallant
				has_trait = generous
				has_trait = gregarious
			}
		}
		modifier = {
			add = -300
			scope:recipient = {
				NOT = { kns_practices_knighthood_trigger = yes }
			}
		}
		modifier = {
			add = -100
			opinion = {
				target = scope:recipient
				value <= low_negative_opinion
			}
		}
	}
}

kns_educate_squire_interaction = {
	category = interaction_category_diplomacy
	desc = kns_educate_squire_interaction_desc
	icon = icon_combat
	use_diplomatic_range = yes
	send_name = KNS_OFFER_EDUCATE_SQUIRE
	greeting = positive

	notification_text = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:actor = scope:secondary_actor
					NOT = { scope:recipient = scope:secondary_recipient }
				}
				desc = KNS_EDUCATE_RECIPIENT_SQUIRE_PERSONALLY_REQUEST
			}
			triggered_desc = {
				trigger = {
					scope:actor = scope:secondary_actor
					scope:recipient = scope:secondary_recipient
				}
				desc = KNS_EDUCATE_SQUIRE_PERSONALLY_REQUEST
			}
			triggered_desc = {
				trigger = {
					NOT = { scope:recipient = scope:secondary_recipient }
					NOT = { scope:actor = scope:secondary_actor }
				}
				desc = KNS_EDUCATE_RECIPIENT_SQUIRE_REQUEST
			}
			desc = KNS_EDUCATE_SQUIRE_REQUEST
		}
	}

	cooldown = { months = 4 }

	redirect = {
		if = {
			limit = { scope:actor = { is_ai = no } }
			scope:actor = {
				if = {
					limit = { kns_is_eligible_for_squirehood_trigger = yes }
					save_scope_as = secondary_recipient
				}
				else_if = {
					limit = { kns_can_take_squire_trigger = yes }
					save_scope_as = secondary_actor
				}
			}
			scope:recipient = {
				if = {
					limit = { kns_can_take_squire_trigger = yes }
					save_scope_as = secondary_actor
				}
				else = {
					save_scope_as = secondary_recipient
				}
			}
		}
	}

	populate_actor_list = {
		scope:actor = {
			if = {
				limit = { kns_can_take_squire_trigger = yes }
				add_to_list = characters
			}
			every_courtier = {
				limit = { kns_can_take_squire_trigger = yes }
				add_to_list = characters
			}
		}
	}

	populate_recipient_list = {
		scope:recipient = {
			if = {
				limit = { kns_is_eligible_for_squirehood_trigger = yes }
				add_to_list = characters
			}
			every_courtier = {
				limit = { kns_is_eligible_for_squirehood_trigger = yes }
				add_to_list = characters
			}
		}
	}

	prompt = {
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = scope:secondary_actor } }
				desc = KNS_EDUCATE_SELECT_YOUR_KNIGHT
			}
			triggered_desc = {
				trigger = { NOT = { exists = scope:secondary_recipient } }
				desc = KNS_EDUCATE_SELECT_YOUR_SQUIRE
			}
		}
	}

	is_shown = {
		scope:recipient = {
			is_in_the_same_court_as = scope:actor
		}
		trigger_if = {
			limit = { exists = scope:secondary_actor }
			scope:secondary_actor = { kns_can_take_squire_trigger = yes }
		}
		trigger_if = {
			limit = { exists = scope:secondary_recipient }
			scope:secondary_recipient = { kns_is_eligible_for_squirehood_trigger = yes }
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_imprisoned = no
			NOT = { has_trait = incapable }
		}
		custom_description = {
			text = kns_character_to_knight_exists
			subject = scope:actor
			scope:actor = {
				OR = {
					any_courtier = { kns_can_take_squire_trigger = yes }
					kns_can_take_squire_trigger = yes
				}
			}
		}
		custom_description = {
			text = kns_character_to_squire_exists
			object = scope:actor
			scope:actor = {
				OR = {
					any_courtier = { kns_is_eligible_for_squirehood_trigger = yes }
					kns_is_eligible_for_squirehood_trigger = yes
				}
			}
		}
	}

	on_accept = {
		scope:secondary_recipient = { save_scope_as = squire }
		scope:secondary_actor = { save_scope_as = knight }
		kns_set_squire_effect = {
			KNIGHT = scope:secondary_actor
			SQUIRE = scope:secondary_recipient
		}
		kns_offer_squire_hook_effect = yes
	}

	auto_accept = yes

	send_option = {
		is_shown = { NOT = { scope:actor = scope:recipient } }
		is_valid = { scope:actor = { has_usable_hook = scope:recipient } }
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"
	send_options_exclusive = no

	ai_frequency = 24

	ai_potential = {
		is_adult = yes
		is_available_ai_adult = yes
		OR = {
			kns_is_knight_trigger = yes
			any_courtier = { kns_can_take_squire_trigger = yes }
		}
		any_courtier = { kns_is_eligible_for_squirehood_trigger = yes }
	}

	ai_targets = { ai_recipients = self }

	ai_will_do = {
		base = 60
		modifier = {
			add = 20
			kns_practices_knighthood_trigger = yes
		}
		modifier = {
			add = {
				value = scope:secondary_recipient.age
				multiply = -4
			}
			scope:secondary_recipient.age >= 16
		}
		modifier = {
			add = -1000
			scope:secondary_recipient = {
				OR = {
					has_relation_rival = scope:secondary_actor
					has_relation_nemesis = scope:secondary_actor
				}
			}
		}
		modifier = {
			factor = 0
			scope:secondary_recipient = {
				NOR = {
					is_close_family_of = scope:actor
					this = scope:actor
				}
			}
		}
	}
}

				desc = KNS_EDUCATE_SELECT_THEIR_SQUIRE
			}
		}
	}

		scope:recipient = {
			NOT = { this = scope:actor }
			is_landed = yes
		}

			limit = { exists = scope:secondary_recipient }

				kns_is_eligible_for_squirehood_trigger = yes
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			text = kns_character_to_squire_exists
			object = scope:recipient
			scope:recipient = {
				OR = {
					any_courtier = { kns_is_eligible_for_squirehood_trigger = yes }
					kns_is_eligible_for_squirehood_trigger = yes
				}
			}
		}
		custom_description = {
			text = kns_character_to_knight_exists
			subject = scope:actor
			scope:actor = {
				OR = {
					any_courtier = { kns_can_take_squire_trigger = yes }
					kns_can_take_squire_trigger = yes
				}
			}
		}
		scope:recipient = { is_imprisoned = no }
		NOT = { scope:actor = { is_at_war_with = scope:recipient } }
	}

	on_accept = {
		scope:secondary_actor = { save_scope_as = knight }
		scope:secondary_recipient = { save_scope_as = squire }
		if = {
			limit = {
				scope:knight = { kns_can_take_squire_trigger = yes }
			}
			kns_set_squire_effect = {
				KNIGHT = scope:secondary_actor
				SQUIRE = scope:secondary_recipient
			}
			kns_offer_squire_hook_effect = yes
			scope:actor = {
				if = {
					limit = { NOT = { this = scope:recipient } }
					trigger_event = kns_interaction.0080
				}
			}
		}
	}

	on_decline = {
		scope:actor = {
			trigger_event = kns_interaction.0081
			add_opinion = {
				modifier = declined_guardianship_opinion
				target = scope:recipient
				opinion = -15
			}
		}
	}

	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:recipient
			scope:hook = yes
			scope:actor = { has_strong_hook = scope:recipient }
		}
	}

	ai_accept = {
		base = 0
		modifier = {
			add = 50
			scope:recipient = { target_is_liege_or_above = scope:actor }
			desc = EDUCATE_CHILD_ACTOR_IS_MY_LIEGE
		}
		modifier = {
			add = 20

		}
		modifier = {
			add = -1000
			scope:recipient = {
				OR = {
					has_relation_rival = scope:actor
					has_relation_nemesis = scope:actor
				}
			}
			desc = AI_YOUR_RIVAL
		}
		opinion_modifier = {
			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 0.8
			desc = AI_OPINION_REASON
		}
		compare_modifier = {
			trigger = { scope:secondary_actor = { prowess >= 10 } }
			target = scope:secondary_actor
			value = prowess
			step = 4
			multiplier = 0.75
			offset = -5
			min = 0
			desc = KNS_AI_GUARDIAN_SKILL_PROWESS
		}
		modifier = {
			add = 100
			scope:actor = { is_allied_to = scope:recipient }
			desc = AI_HAS_ALLIANCE
		}
	}

	send_option = {
		is_shown = { NOT = { scope:actor = scope:recipient } }
		is_valid = { scope:actor = { has_usable_hook = scope:recipient } }
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"
	send_options_exclusive = no

	ai_targets = { ai_recipients = vassals ai_recipients = peer_vassals }
	ai_frequency = 36

	ai_potential = {
		is_imprisoned = no
		is_available = yes
		any_courtier = { kns_can_take_squire_trigger = yes }
	}

	ai_will_do = {

		modifier = {

			AND = {
				scope:recipient = { kns_practices_knighthood_trigger = yes }
				scope:actor = { kns_practices_knighthood_trigger = yes }

				OR = {
					has_relation_rival = scope:actor
					has_relation_nemesis = scope:actor

			}
		}
	}
