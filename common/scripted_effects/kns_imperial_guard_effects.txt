kns_unequip_imperial_guard_kit_slots_effect = {
	every_equipped_character_artifact = {
		limit = {
			OR = {
				artifact_slot_type = primary_armament
				artifact_slot_type = armor
			}
		}
		add_to_list = kns_ig_kit_unequip
	}
	every_in_list = {
		list = kns_ig_kit_unequip
		unequip_artifact_from_owner = yes
	}
}

kns_strip_extra_equipped_weapons_effect = {
	every_equipped_character_artifact = {
		limit = {
			artifact_slot_type = primary_armament
			exists = scope:kns_ig_new_weapon
			NOT = { this = scope:kns_ig_new_weapon }
		}
		add_to_list = kns_ig_extra_weapons
	}
	every_in_list = {
		list = kns_ig_extra_weapons
		unequip_artifact_from_owner = yes
	}
}

kns_force_equip_created_artifact_to_kit_owner_effect = {
	save_scope_as = kns_ig_to_equip
	scope:kns_kit_owner = {
		if = {
			limit = { scope:kns_ig_to_equip = { artifact_slot_type = primary_armament } }
			every_equipped_character_artifact = {
				limit = { artifact_slot_type = primary_armament }
				add_to_list = kns_ig_force_unequip
			}
		}
		else_if = {
			limit = { scope:kns_ig_to_equip = { artifact_slot_type = armor } }
			every_equipped_character_artifact = {
				limit = { artifact_slot_type = armor }
				add_to_list = kns_ig_force_unequip
			}
		}
		every_in_list = {
			list = kns_ig_force_unequip
			unequip_artifact_from_owner = yes
		}
	}
	if = {
		limit = {
			scope:kns_kit_owner = { can_equip_artifact = scope:kns_ig_to_equip }
		}
		equip_artifact_to_owner = yes
	}
	else = {
		equip_artifact_to_owner_replace = yes
	}
}

		kns_ig_give_accolade_title_nickname_effect = yes

			remove_trait = squire

			add_trait = knight

	}
}

			NOT = { is_in_list = kns_ig_cand_pool }
		}

	}
	every_knight = {
		limit = {
			kns_can_be_appointed_imperial_guard_by_trigger = { EMPEROR = scope:kns_ig_emperor }
			NOT = { is_in_list = kns_ig_cand_pool }
		}
		add_to_list = kns_ig_cand_pool
	}
	every_vassal_or_below = {
		limit = { is_alive = yes }
		if = {
			limit = {
				kns_can_be_appointed_imperial_guard_by_trigger = { EMPEROR = scope:kns_ig_emperor }
				NOT = { is_in_list = kns_ig_cand_pool }
			}

		}

				kns_can_be_appointed_imperial_guard_by_trigger = { EMPEROR = scope:kns_ig_emperor }
				NOT = { is_in_list = kns_ig_cand_pool }
			}
			add_to_list = kns_ig_cand_pool
		}
	}
	if = {
		limit = { exists = var:kns_ig_sort_potential }
		ordered_in_list = {
			list = kns_ig_cand_pool
			limit = { has_variable = kns_prowess_cap }
			order_by = {
				value = 0
				add = var:kns_prowess_cap
			}
			max = 50
			check_range_bounds = no
			save_scope_as = kns_ig_cand
			scope:kns_ig_emperor = {
				add_to_variable_list = {
					name = kns_imperial_guard_candidates
					target = scope:kns_ig_cand
				}
			}
		}
	}
	else = {
		ordered_in_list = {
			list = kns_ig_cand_pool
			order_by = prowess
			max = 50
			check_range_bounds = no
			save_scope_as = kns_ig_cand
			scope:kns_ig_emperor = {
				add_to_variable_list = {

					target = scope:kns_ig_cand
				}
			}
		}
	}

	clear_variable_list = kns_imperial_guard_candidates
	kns_clear_imperial_guard_slot_variables_effect = yes
	set_variable = {
		name = kns_ig_guard_count
		value = 0
	}
	every_relation = {
		type = kns_imperial_guard
		limit = { is_alive = yes }
		save_scope_as = kns_ig_member
		scope:kns_ig_emperor = {
			add_to_variable_list = {
				name = kns_imperial_guards
				target = scope:kns_ig_member
			}
			change_variable = {
				name = kns_ig_guard_count
				add = 1
			}
		}
		scope:kns_ig_member = {
			kns_infer_imperial_guard_post_from_weapon_effect = yes
			kns_bind_imperial_guard_post_effect = yes
			kns_ig_adopt_equipped_kit_if_unbound_effect = {
				EMPEROR = scope:kns_ig_emperor
			}
			kns_ig_clear_accolade_successor_status_effect = {
				KNIGHT = scope:kns_ig_member
				EMPEROR = scope:kns_ig_emperor
			}
		}
	}
	if = {
		limit = { exists = var:kns_ig_hire_window_open }
		kns_ig_collect_imperial_guard_candidates_effect = yes
	}
	if = {
		limit = {
			exists = var:kns_ig_attire_target
			OR = {
				var:kns_ig_attire_target = { is_alive = no }
				NOT = {
					var:kns_ig_attire_target = {
						kns_is_imperial_guard_of_trigger = { EMPEROR = scope:kns_ig_emperor }
					}
				}
			}
		}
		remove_variable = kns_ig_attire_target
		remove_variable = kns_ig_attire_window_open
	}
}

kns_sweep_invalid_imperial_guards_effect = {
	if = {
		limit = {
			num_of_relation_kns_imperial_guard > 0
			OR = {
				NOT = { kns_is_emperor_trigger = yes }
				NOT = { kns_practices_knighthood_trigger = yes }
			}
		}
		kns_dismiss_all_imperial_guards_effect = yes
	}
	else = {
		every_relation = {
			type = kns_imperial_guard
			limit = {
				OR = {
					is_alive = no
					AND = {
						is_landed = yes
						NOT = { kns_ig_may_remain_landed_guard_trigger = yes }
					}
					AND = {
						is_landed = no
						NOT = { is_courtier_of = prev }
					}
					NOT = { kns_practices_knighthood_trigger = yes }
				}
			}
			add_to_list = kns_ig_invalid
		}
		every_in_list = {
			list = kns_ig_invalid
			kns_dismiss_imperial_guard_effect = { KNIGHT = this }
		}
		every_relation = {
			type = kns_imperial_guard
			limit = { is_alive = yes }
			kns_ig_check_guard_kit_in_wrong_hands_effect = yes
			kns_ig_restore_knight_kit_durability_effect = yes
		}
		kns_ig_restore_emperor_post_kit_durability_effect = yes
		kns_ig_clear_all_imperial_guard_successors_effect = yes
	}
}
