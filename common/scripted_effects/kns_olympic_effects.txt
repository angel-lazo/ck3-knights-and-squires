kns_schedule_next_olympic_effect = {

		limit = { NOT = { has_variable = kns_olympic_interval_years } }
		set_variable = {
			name = kns_olympic_interval_years
			value = 4
		}
	}
	if = {
		limit = { NOT = { has_variable = kns_olympic_prize_gold } }

			name = kns_olympic_prize_gold

		}

	set_variable = {
		name = kns_olympic_next_year
		value = current_year
	}
	change_variable = {
		name = kns_olympic_next_year
		add = var:kns_olympic_interval_years
	}
}

		clear_variable_list = $LIST$

		$TARGET$ = {
			add_to_variable_list = {
				name = $LIST$
				target = scope:kns_olympic_copy_item
			}
		}
	}
}

kns_olympic_sync_push_winner_effect = {
	$WINNER$ = {
		if = {
			limit = { NOT = { has_variable = kns_imperial_duel_wins } }
			set_variable = {
				name = kns_imperial_duel_wins
				value = 0
			}
		}
		change_variable = {
			name = kns_imperial_duel_wins
			add = 1
		}
	}
	if = {
		limit = { exists = scope:kns_olympic_sync_loser }
		kns_olympic_create_bout_memories_effect = {
			WINNER = $WINNER$
			LOSER = scope:kns_olympic_sync_loser
		}
		scope:kns_olympic_sync_loser = {
			if = {
				limit = { is_alive = yes }
				if = {
					limit = { NOT = { has_variable = kns_imperial_duel_losses } }
					set_variable = {
						name = kns_imperial_duel_losses
						value = 0
					}
				}
				change_variable = {
					name = kns_imperial_duel_losses
					add = 1
				}
			}
		}
	}
	kns_register_imperial_hof_character_effect = { CHAR = $WINNER$ }
	if = {
		limit = {
			exists = scope:kns_olympic_sync_loser
			scope:kns_olympic_sync_loser = { is_alive = yes }
		}
		kns_register_imperial_hof_character_effect = { CHAR = scope:kns_olympic_sync_loser }
	}
	add_to_variable_list = {
		name = kns_olympic_next_round
		target = $WINNER$
	}
	remove_variable = kns_olympic_bout_open
}

kns_olympic_force_complete_bracket_effect = {
	save_scope_as = kns_emperor
	kns_olympic_debug_log_effect = { MSG = force_complete_on_organizer_death }
	if = {
		limit = {
			has_variable = kns_olympic_bout_open
			exists = var:kns_olympic_fighter_a
			exists = var:kns_olympic_fighter_b
		}
		var:kns_olympic_fighter_a = { save_scope_as = kns_olympic_fighter_a }
		var:kns_olympic_fighter_b = { save_scope_as = kns_olympic_fighter_b }
		random_list = {
			5 = {
				modifier = {
					add = scope:kns_olympic_fighter_a.kns_olympic_fake_duel_prowess_weight_value
				}
				scope:kns_olympic_fighter_b = { save_scope_as = kns_olympic_sync_loser }
				kns_olympic_sync_push_winner_effect = { WINNER = scope:kns_olympic_fighter_a }
			}
			5 = {
				modifier = {
					add = scope:kns_olympic_fighter_b.kns_olympic_fake_duel_prowess_weight_value
				}
				scope:kns_olympic_fighter_a = { save_scope_as = kns_olympic_sync_loser }
				kns_olympic_sync_push_winner_effect = { WINNER = scope:kns_olympic_fighter_b }
			}
		}
	}
	while = {
		count = 250
		limit = {
			OR = {
				variable_list_size = {
					name = kns_olympic_round_fighters
					value > 0
				}
				variable_list_size = {
					name = kns_olympic_next_round
					value > 1
				}
			}
		}
		if = {
			limit = {
				variable_list_size = {
					name = kns_olympic_round_fighters
					value >= 2
				}
			}
			kns_olympic_draw_bout_pair_from_round_effect = yes
			if = {
				limit = {
					var:kns_olympic_found_a = 1
					var:kns_olympic_found_b = 1
				}
				random_list = {
					5 = {
						modifier = {
							add = scope:kns_olympic_fighter_a.kns_olympic_fake_duel_prowess_weight_value
						}
						scope:kns_olympic_fighter_b = { save_scope_as = kns_olympic_sync_loser }
						kns_olympic_sync_push_winner_effect = { WINNER = scope:kns_olympic_fighter_a }
					}
					5 = {
						modifier = {
							add = scope:kns_olympic_fighter_b.kns_olympic_fake_duel_prowess_weight_value
						}
						scope:kns_olympic_fighter_a = { save_scope_as = kns_olympic_sync_loser }
						kns_olympic_sync_push_winner_effect = { WINNER = scope:kns_olympic_fighter_b }
					}
				}
			}
			else_if = {
				limit = { var:kns_olympic_found_a = 1 }
				kns_olympic_sync_push_winner_effect = { WINNER = scope:kns_olympic_fighter_a }
			}
			else = {
				clear_variable_list = kns_olympic_round_fighters
			}
		}
		else_if = {
			limit = {
				variable_list_size = {
					name = kns_olympic_round_fighters
					value = 1
				}
			}
			random_in_list = {
				variable = kns_olympic_round_fighters
				limit = { is_alive = yes }
				save_scope_as = kns_olympic_bye_winner
			}
			remove_list_variable = {
				name = kns_olympic_round_fighters
				target = scope:kns_olympic_bye_winner
			}
			if = {
				limit = { exists = scope:kns_olympic_bye_winner }
				add_to_variable_list = {
					name = kns_olympic_next_round
					target = scope:kns_olympic_bye_winner
				}
			}
		}
		else_if = {
			limit = {
				variable_list_size = {
					name = kns_olympic_round_fighters
					value = 0
				}
				variable_list_size = {
					name = kns_olympic_next_round
					value > 1
				}
			}
			every_in_list = {
				variable = kns_olympic_next_round
				limit = { is_alive = yes }
				save_scope_as = kns_olympic_advancer
				scope:kns_emperor = {
					add_to_variable_list = {
						name = kns_olympic_round_fighters
						target = scope:kns_olympic_advancer
					}
				}
			}
			clear_variable_list = kns_olympic_next_round
			if = {
				limit = { has_variable = kns_olympic_round }
				change_variable = {
					name = kns_olympic_round
					add = 1
				}
			}
		}
	}
	random_in_list = {
		variable = kns_olympic_next_round
		limit = { is_alive = yes }
		save_scope_as = kns_olympic_winner
	}
	if = {
		limit = { NOT = { exists = scope:kns_olympic_winner } }
		random_in_list = {
			variable = kns_olympic_round_fighters
			limit = { is_alive = yes }
			save_scope_as = kns_olympic_winner
		}
	}
	if = {
		limit = { exists = scope:kns_olympic_winner }
		kns_award_imperial_olympic_winner_effect = { WINNER = scope:kns_olympic_winner }
		if = {
			limit = {
				exists = scope:kns_olympic_winner
				scope:kns_olympic_winner = { is_ai = no }
			}
			kns_olympic_store_champion_scope_effect = yes
			scope:kns_olympic_winner = {
				trigger_event = kns_imperial_olympic.0009
			}
		}
	}
	kns_clear_imperial_olympic_vars_effect = yes
	kns_olympic_store_champion_scope_effect = yes
}

kns_handle_imperial_olympic_organizer_death_effect = {
	if = {
		limit = { kns_has_olympic_settings_to_save_trigger = yes }
		kns_olympic_debug_log_effect = { MSG = organizer_death }
		kns_merge_character_hof_index_into_global_effect = yes
		kns_save_olympic_dormant_from_this_effect = yes
		if = {
			limit = { exists = player_heir }
			player_heir = {
				save_scope_as = kns_olympic_successor
			}
		}
		else_if = {
			limit = { exists = primary_heir }
			primary_heir = {
				save_scope_as = kns_olympic_successor
			}
		}
		if = {
			limit = { has_character_flag = kns_imperial_olympic_host }
			if = {
				limit = {
					is_ai = no
					exists = scope:kns_olympic_successor
					scope:kns_olympic_successor = { is_alive = yes }
				}
				kns_copy_olympic_runtime_to_effect = { TARGET = scope:kns_olympic_successor }
				kns_clear_imperial_olympic_vars_effect = yes
				scope:kns_olympic_successor = {
					save_scope_as = kns_emperor
					if = {
						limit = { has_variable = kns_olympic_bout_open }
						trigger_event = kns_imperial_olympic.0102
					}
					else = {
						trigger_event = {
							id = kns_imperial_olympic.0100
							days = 1
						}
					}
					trigger_event = kns_imperial_olympic.0031
				}
			}
			else = {
				kns_olympic_force_complete_bracket_effect = yes
			}
		}
	}
}

	}

		variable = kns_olympic_player_champion_picks

			scope:kns_emperor = {

					target = prev
				}
				variable_list_size = {

					value < var:kns_olympic_bracket_size
				}
			}
		}
		save_scope_as = kns_olympic_seed_pick
		scope:kns_emperor = {
			remove_list_variable = {
				name = kns_olympic_entrants

				name = kns_olympic_round_fighters
				target = scope:kns_olympic_seed_pick
			}

	}
	while = {
		count = 256
		limit = {
			variable_list_size = {
				name = kns_olympic_entrants
				value > 0
			}
			variable_list_size = {
				name = kns_olympic_round_fighters
				value < var:kns_olympic_bracket_size
			}
		}
		set_variable = {
			name = kns_olympic_found_seed
			value = 0
		}
		random_in_list = {
			variable = kns_olympic_entrants
			limit = {
				is_alive = yes
				kns_can_enter_imperial_olympic_duel_trigger = yes
			}
			weight = {
				base = 1
				modifier = {
					add = prowess_no_portrait
				}
			}
			save_scope_as = kns_olympic_seed_pick
			scope:kns_emperor = {
				set_variable = {
					name = kns_olympic_found_seed
					value = 1
				}
			}
		}
		if = {
			limit = {
				var:kns_olympic_found_seed = 1
				exists = scope:kns_olympic_seed_pick
			}
			remove_list_variable = {
				name = kns_olympic_entrants
				target = scope:kns_olympic_seed_pick
			}
			add_to_variable_list = {
				name = kns_olympic_round_fighters
				target = scope:kns_olympic_seed_pick
			}
		}
		else = {
			clear_variable_list = kns_olympic_entrants
		}
	}
}

# Survive bracket teardown and leftover combat-scope wipes until the crown event is dismissed.
# save_scope_as first: nested set_variable value = scope:winner can silently store the host.
kns_olympic_store_champion_scope_effect = {
	if = {
		limit = { exists = scope:kns_olympic_winner }
		scope:kns_olympic_winner = {
			save_scope_as = kns_olympic_champion_char
		}
		if = {
			limit = { NOT = { exists = scope:kns_emperor } }
			kns_olympic_save_host_effect = yes
		}
		if = {
			limit = { exists = scope:kns_emperor }
			scope:kns_emperor = {
				set_variable = {
					name = kns_olympic_champion
					value = scope:kns_olympic_champion_char
				}
			}
		}
		else = {
			set_variable = {
				name = kns_olympic_champion
				value = scope:kns_olympic_champion_char
			}
		}
	}
}

# Always overwrite winner from the stored character. Portraits of the visible crown
# event are built before that event's immediate, so restore in a hidden setup event first.
kns_olympic_restore_champion_scope_effect = {
	if = {
		limit = { NOT = { exists = scope:kns_emperor } }
		kns_olympic_save_host_effect = yes
	}
	if = {
		limit = {
			NOT = { exists = scope:kns_emperor }
			exists = var:kns_olympic_champion
		}
		save_scope_as = kns_emperor
	}
	if = {
		limit = {
			NOT = { exists = scope:kns_emperor }
		}
		every_player = {
			limit = { exists = var:kns_olympic_champion }
			save_scope_as = kns_emperor
		}
	}
	if = {
		limit = {
			exists = scope:kns_emperor
			scope:kns_emperor = { exists = var:kns_olympic_champion }
		}
		scope:kns_emperor = {
			var:kns_olympic_champion = {
				save_scope_as = kns_olympic_winner
			}
		}
	}
	else_if = {
		limit = { exists = var:kns_olympic_champion }
		var:kns_olympic_champion = {
			save_scope_as = kns_olympic_winner
		}
	}
}

kns_olympic_extract_human_pair_from_round_effect = {
	remove_variable = kns_olympic_player_pair_pending
	remove_variable = kns_olympic_pending_a
	remove_variable = kns_olympic_pending_b
	if = {
		limit = { exists = scope:kns_olympic_human_fighter }
		clear_saved_scope = kns_olympic_human_fighter
	}
	if = {
		limit = { exists = scope:kns_olympic_human_foe }
		clear_saved_scope = kns_olympic_human_foe
	}
	set_variable = {
		name = kns_olympic_found_human
		value = 0
	}
	set_variable = {
		name = kns_olympic_found_foe
		value = 0
	}
	random_in_list = {
		variable = kns_olympic_round_fighters
		limit = {
			is_alive = yes
			is_ai = no
		}
		save_scope_as = kns_olympic_human_fighter
		scope:kns_emperor = {
			set_variable = {
				name = kns_olympic_found_human
				value = 1
			}
		}
	}
	if = {
		limit = { var:kns_olympic_found_human = 1 }
		remove_list_variable = {
			name = kns_olympic_round_fighters
			target = scope:kns_olympic_human_fighter
		}
		random_in_list = {
			variable = kns_olympic_round_fighters
			limit = { is_alive = yes }
			save_scope_as = kns_olympic_human_foe
			scope:kns_emperor = {
				set_variable = {
					name = kns_olympic_found_foe
					value = 1
				}
			}
		}
		if = {
			limit = { var:kns_olympic_found_foe = 1 }
			remove_list_variable = {
				name = kns_olympic_round_fighters
				target = scope:kns_olympic_human_foe
			}
			set_variable = {
				name = kns_olympic_pending_a
				value = scope:kns_olympic_human_fighter
			}
			set_variable = {
				name = kns_olympic_pending_b
				value = scope:kns_olympic_human_foe
			}
			set_variable = kns_olympic_player_pair_pending
			kns_olympic_debug_log_effect = { MSG = extract_player_pair }
		}
		else = {
			kns_olympic_debug_log_effect = { MSG = extract_player_bye }
			kns_record_imperial_olympic_bye_advance_effect = {
				WINNER = scope:kns_olympic_human_fighter
			}
		}
	}
	else = {
		kns_olympic_debug_log_effect = { MSG = extract_no_player }
	}
	remove_variable = kns_olympic_found_human
	remove_variable = kns_olympic_found_foe
}

			base = 1

			}
		}
		else_if = {
			limit = {
				variable_list_size = {
					name = kns_olympic_next_round
					value = 0
				}
			}

			kns_clear_imperial_olympic_vars_effect = yes

			kns_olympic_debug_log_effect = { MSG = end_round_advance }

				variable = kns_olympic_next_round
				limit = { is_alive = yes }

				scope:kns_emperor = {
					add_to_variable_list = {
						name = kns_olympic_round_fighters
						target = scope:kns_olympic_advancer
					}
				}
			}
			clear_variable_list = kns_olympic_next_round
			change_variable = {
				name = kns_olympic_round
				add = 1
			}

				limit = {

						name = kns_olympic_round_fighters
						value = 0
					}
				}
				trigger_event = kns_imperial_olympic.0005
				kns_clear_imperial_olympic_vars_effect = yes
			}
		}
	}
}

kns_award_imperial_olympic_winner_effect = {
	$WINNER$ = {
		save_scope_as = kns_olympic_winner
		add_prestige = 5000
		create_character_memory = {
			type = kns_olympic_won_the_games
		}
		if = {
			limit = { NOT = { has_variable = kns_imperial_titles_won } }
			set_variable = {
				name = kns_imperial_titles_won
				value = 0
			}
		}
		change_variable = {
			name = kns_imperial_titles_won
			add = 1
		}
		kns_record_olympic_tournament_year_effect = yes
	}
	if = {
		limit = {
			scope:kns_emperor = {
				has_variable = kns_olympic_prize_gold
				var:kns_olympic_prize_gold >= 1000
			}
		}
		scope:kns_emperor = {
			add_gold = {
				value = var:kns_olympic_prize_gold
				multiply = -1
			}
		}
		$WINNER$ = {
			add_gold = scope:kns_emperor.kns_olympic_winner_gold_value
			if = {
				limit = { exists = dynasty }
				add_dynasty_prestige = scope:kns_emperor.kns_olympic_winner_renown_value
			}
		}
	}
	if = {
		limit = {
			NOT = { $WINNER$ = scope:kns_emperor }
			$WINNER$ = { kns_is_knight_trigger = no }
		}
		kns_become_a_knight_effect = {
			KNIGHT_TO_BE = $WINNER$
			KNIGHT_MAKER = scope:kns_emperor
		}
	}
	scope:kns_emperor = {
		kns_apply_olympic_knight_culture_modifier_effect = yes
		kns_apply_olympic_host_county_bonus_effect = yes
	}
	kns_register_imperial_hof_character_effect = { CHAR = $WINNER$ }
	$WINNER$ = {
		add_opinion = {
			target = scope:kns_emperor
			modifier = kns_olympic_champion_opinion
		}
	}
}

		root = {

				name = kns_olympic_picker_count
				add = 1
			}
		}

}

kns_rebuild_olympic_champion_picker_list_effect = {
	save_scope_as = kns_emperor
	clear_variable_list = kns_olympic_eligible_champions
	if = {
		limit = { kns_olympic_fit_to_fight_trigger = yes }
		add_to_variable_list = {
			name = kns_olympic_eligible_champions
			target = scope:kns_emperor
		}
	}
	every_courtier = {
		limit = {
			kns_can_pick_as_olympic_champion_trigger = yes
		}
		add_to_list = kns_olympic_picker_unsorted
	}
	ordered_in_list = {
		list = kns_olympic_picker_unsorted
		order_by = prowess_no_portrait
		max = 100
		check_range_bounds = no
		save_temporary_scope_as = kns_olympic_picker_candidate
		scope:kns_emperor = {
			add_to_variable_list = {
				name = kns_olympic_eligible_champions
				target = scope:kns_olympic_picker_candidate
			}
		}
	}
}

kns_start_imperial_olympic_player_bout_effect = {
	kns_restore_imperial_olympic_bout_scopes_effect = yes
	if = {
		limit = {
			exists = scope:kns_olympic_fighter_a
			exists = scope:kns_olympic_fighter_b
			NOT = { scope:kns_olympic_fighter_a = scope:kns_olympic_fighter_b }
		}
		if = {
			limit = {
				NOT = { exists = scope:kns_olympic_player_combatant }
			}
			kns_olympic_assign_minigame_sides_effect = yes
		}
		if = {
			limit = {
				exists = scope:kns_olympic_player_combatant
				exists = scope:kns_olympic_ai_combatant
			}
			hidden_effect = {
				kns_olympic_reset_vanilla_duel_state_effect = yes
				# Do not pre-set sc_victor/sc_loser: leftover sc_finished then instant-ends the next duel.
				kns_olympic_debug_log_bout_effect = { MSG = player_bout_start_before_combat }
				configure_start_single_combat_effect = {
					SC_INITIATOR = scope:kns_olympic_player_combatant
					SC_ATTACKER = scope:kns_olympic_ai_combatant
					SC_DEFENDER = scope:kns_olympic_player_combatant
					FATALITY = practice
					FIXED = no
					LOCALE = courtyard
					OUTPUT_EVENT = kns_imperial_olympic.0103
					INVALIDATION_EVENT = kns_imperial_olympic.0105
				}
				if = {
					limit = { scope:kns_olympic_player_combatant = { is_ai = yes } }
					kns_olympic_schedule_knight_bout_hang_watch_effect = yes
				}
				else = {
					remove_variable = kns_olympic_hang_watch
				}
				kns_olympic_debug_log_bout_effect = { MSG = player_bout_start_ok }
			}
		}
		else = {
			kns_olympic_debug_log_bout_effect = { MSG = player_bout_start_no_player }
		}
	}
	else = {
		kns_olympic_debug_log_bout_effect = { MSG = player_bout_start_bad_pair }
	}
}

kns_repair_olympic_low_durability_gear_effect = {
	every_equipped_character_artifact = {
		limit = {
			OR = {
				artifact_slot_type = primary_armament
				artifact_slot_type = armor
			}
			artifact_durability_percent <= 0.15
		}
		add_durability = {
			value = artifact_max_durability
			subtract = artifact_durability
		}
	}
}
