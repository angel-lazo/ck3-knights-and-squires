# Runtime only. Keeps the institution: interval, purse, bracket size, venue, next year.
kns_clear_imperial_olympic_vars_effect = {
	remove_character_flag = kns_imperial_olympic_host
	remove_character_flag = kns_entered_imperial_olympic
	remove_character_flag = kns_olympic_manage_session
	remove_character_flag = kns_olympic_picking_venue
	remove_variable = kns_olympic_invited_ruler_count
	remove_variable = kns_olympic_slots_per_ruler
	remove_variable = kns_olympic_venue_walk
	remove_variable = kns_olympic_round
	remove_variable = kns_olympic_fighter_a
	remove_variable = kns_olympic_fighter_b
	remove_variable = kns_olympic_bout_open
	remove_variable = kns_olympic_found_seed
	clear_variable_list = kns_olympic_entrants
	clear_variable_list = kns_olympic_invited_rulers
	clear_variable_list = kns_olympic_round_fighters
	clear_variable_list = kns_olympic_next_round
	clear_variable_list = kns_olympic_venue_scratch
}

kns_abandon_imperial_olympic_planning_effect = {
	kns_clear_imperial_olympic_vars_effect = yes
	if = {
		limit = { NOT = { kns_has_olympic_institution_trigger = yes } }
		remove_variable = kns_olympic_bracket_size
		remove_variable = kns_olympic_location
		remove_variable = kns_olympic_interval_years
		remove_variable = kns_olympic_prize_gold
		remove_variable = kns_olympic_next_year
	}
}

kns_olympic_continue_setup_or_manage_effect = {
	if = {
		limit = { has_character_flag = kns_olympic_manage_session }
		trigger_event = kns_imperial_olympic.0020
	}
	else = {
		trigger_event = $NEXT$
	}
}

kns_schedule_next_olympic_effect = {
	if = {
		limit = { NOT = { has_variable = kns_olympic_interval_years } }
		set_variable = {
			name = kns_olympic_interval_years
			value = 4
		}
	}
	if = {
		limit = { NOT = { has_variable = kns_olympic_prize_gold } }
		set_variable = {
			name = kns_olympic_prize_gold
			value = 1000
		}
	}
	set_variable = {
		name = kns_olympic_next_year
		value = current_year
	}
	change_variable = {
		name = kns_olympic_next_year
		add = var:kns_olympic_interval_years
	}
}

kns_save_olympic_dormant_from_this_effect = {
	set_global_variable = kns_olympic_dormant
	set_global_variable = {
		name = kns_olympic_dormant_predecessor
		value = this
	}
	if = {
		limit = { has_variable = kns_olympic_interval_years }
		set_global_variable = {
			name = kns_olympic_dormant_interval
			value = var:kns_olympic_interval_years
		}
	}
	if = {
		limit = { has_variable = kns_olympic_prize_gold }
		set_global_variable = {
			name = kns_olympic_dormant_prize
			value = var:kns_olympic_prize_gold
		}
	}
	if = {
		limit = { has_variable = kns_olympic_bracket_size }
		set_global_variable = {
			name = kns_olympic_dormant_bracket
			value = var:kns_olympic_bracket_size
		}
	}
	if = {
		limit = { has_variable = kns_olympic_next_year }
		set_global_variable = {
			name = kns_olympic_dormant_next_year
			value = var:kns_olympic_next_year
		}
	}
	if = {
		limit = { exists = var:kns_olympic_location }
		set_global_variable = {
			name = kns_olympic_dormant_location
			value = var:kns_olympic_location
		}
	}
}

kns_copy_olympic_list_to_effect = {
	$TARGET$ = {
		clear_variable_list = $LIST$
	}
	every_in_list = {
		variable = $LIST$
		save_scope_as = kns_olympic_copy_item
		$TARGET$ = {
			add_to_variable_list = {
				name = $LIST$
				target = scope:kns_olympic_copy_item
			}
		}
	}
}

kns_olympic_sync_push_winner_effect = {
	$WINNER$ = {
		if = {
			limit = { NOT = { has_variable = kns_imperial_duel_wins } }
			set_variable = {
				name = kns_imperial_duel_wins
				value = 0
			}
		}
		change_variable = {
			name = kns_imperial_duel_wins
			add = 1
		}
	}
	if = {
		limit = { exists = scope:kns_olympic_sync_loser }
		$WINNER$ = {
			create_character_memory = {
				type = kns_olympic_defeated_in_lists
				participants = {
					opponent = scope:kns_olympic_sync_loser
				}
			}
		}
	}
	kns_register_imperial_hof_character_effect = { CHAR = $WINNER$ }
	add_to_variable_list = {
		name = kns_olympic_next_round
		target = $WINNER$
	}
	remove_variable = kns_olympic_bout_open
}

kns_olympic_force_complete_bracket_effect = {
	save_scope_as = kns_emperor
	kns_olympic_debug_log_effect = { MSG = force_complete_on_organizer_death }
	if = {
		limit = {
			has_variable = kns_olympic_bout_open
			exists = var:kns_olympic_fighter_a
			exists = var:kns_olympic_fighter_b
		}
		var:kns_olympic_fighter_a = { save_scope_as = kns_olympic_fighter_a }
		var:kns_olympic_fighter_b = { save_scope_as = kns_olympic_fighter_b }
		random_list = {
			50 = {
				modifier = {
					add = scope:kns_olympic_fighter_a.prowess_no_portrait
				}
				scope:kns_olympic_fighter_b = { save_scope_as = kns_olympic_sync_loser }
				kns_olympic_sync_push_winner_effect = { WINNER = scope:kns_olympic_fighter_a }
			}
			50 = {
				modifier = {
					add = scope:kns_olympic_fighter_b.prowess_no_portrait
				}
				scope:kns_olympic_fighter_a = { save_scope_as = kns_olympic_sync_loser }
				kns_olympic_sync_push_winner_effect = { WINNER = scope:kns_olympic_fighter_b }
			}
		}
	}
	while = {
		count = 250
		limit = {
			OR = {
				variable_list_size = {
					name = kns_olympic_round_fighters
					value > 0
				}
				variable_list_size = {
					name = kns_olympic_next_round
					value > 1
				}
			}
		}
		if = {
			limit = {
				variable_list_size = {
					name = kns_olympic_round_fighters
					value >= 2
				}
			}
			random_in_list = {
				variable = kns_olympic_round_fighters
				limit = { is_alive = yes }
				save_scope_as = kns_olympic_fighter_a
			}
			if = {
				limit = { exists = scope:kns_olympic_fighter_a }
				remove_list_variable = {
					name = kns_olympic_round_fighters
					target = scope:kns_olympic_fighter_a
				}
				random_in_list = {
					variable = kns_olympic_round_fighters
					limit = {
						is_alive = yes
						NOT = { this = scope:kns_olympic_fighter_a }
					}
					save_scope_as = kns_olympic_fighter_b
				}
				if = {
					limit = { exists = scope:kns_olympic_fighter_b }
					remove_list_variable = {
						name = kns_olympic_round_fighters
						target = scope:kns_olympic_fighter_b
					}
					random_list = {
						50 = {
							modifier = {
								add = scope:kns_olympic_fighter_a.prowess_no_portrait
							}
							scope:kns_olympic_fighter_b = { save_scope_as = kns_olympic_sync_loser }
							kns_olympic_sync_push_winner_effect = { WINNER = scope:kns_olympic_fighter_a }
						}
						50 = {
							modifier = {
								add = scope:kns_olympic_fighter_b.prowess_no_portrait
							}
							scope:kns_olympic_fighter_a = { save_scope_as = kns_olympic_sync_loser }
							kns_olympic_sync_push_winner_effect = { WINNER = scope:kns_olympic_fighter_b }
						}
					}
				}
				else = {
					kns_olympic_sync_push_winner_effect = { WINNER = scope:kns_olympic_fighter_a }
				}
			}
		}
		else_if = {
			limit = {
				variable_list_size = {
					name = kns_olympic_round_fighters
					value = 1
				}
			}
			random_in_list = {
				variable = kns_olympic_round_fighters
				limit = { is_alive = yes }
				save_scope_as = kns_olympic_bye_winner
			}
			remove_list_variable = {
				name = kns_olympic_round_fighters
				target = scope:kns_olympic_bye_winner
			}
			if = {
				limit = { exists = scope:kns_olympic_bye_winner }
				add_to_variable_list = {
					name = kns_olympic_next_round
					target = scope:kns_olympic_bye_winner
				}
			}
		}
		else_if = {
			limit = {
				variable_list_size = {
					name = kns_olympic_round_fighters
					value = 0
				}
				variable_list_size = {
					name = kns_olympic_next_round
					value > 1
				}
			}
			every_in_list = {
				variable = kns_olympic_next_round
				limit = { is_alive = yes }
				save_scope_as = kns_olympic_advancer
				scope:kns_emperor = {
					add_to_variable_list = {
						name = kns_olympic_round_fighters
						target = scope:kns_olympic_advancer
					}
				}
			}
			clear_variable_list = kns_olympic_next_round
			if = {
				limit = { has_variable = kns_olympic_round }
				change_variable = {
					name = kns_olympic_round
					add = 1
				}
			}
		}
	}
	random_in_list = {
		variable = kns_olympic_next_round
		limit = { is_alive = yes }
		save_scope_as = kns_olympic_winner
	}
	if = {
		limit = { NOT = { exists = scope:kns_olympic_winner } }
		random_in_list = {
			variable = kns_olympic_round_fighters
			limit = { is_alive = yes }
			save_scope_as = kns_olympic_winner
		}
	}
	if = {
		limit = { exists = scope:kns_olympic_winner }
		kns_award_imperial_olympic_winner_effect = { WINNER = scope:kns_olympic_winner }
		if = {
			limit = {
				exists = scope:kns_olympic_winner
				scope:kns_olympic_winner = { is_ai = no }
			}
			scope:kns_olympic_winner = {
				trigger_event = kns_imperial_olympic.0011
			}
		}
	}
	kns_clear_imperial_olympic_vars_effect = yes
}

	add_character_flag = kns_olympic_institution
	if = {
		limit = { NOT = { has_variable = kns_olympic_interval_years } }
		set_variable = {
			name = kns_olympic_interval_years
			value = 4
		}
	}
	if = {

			value = 1000
		}
	}
	remove_character_flag = kns_olympic_continue_session
	kns_apply_olympic_knight_culture_modifier_effect = yes
	kns_clear_olympic_dormant_effect = yes
}

kns_olympic_after_prize_setup_effect = {
	if = {
		limit = { has_character_flag = kns_olympic_manage_session }
		trigger_event = kns_imperial_olympic.0020
	}
	else_if = {
		limit = { has_character_flag = kns_olympic_continue_session }
		kns_adopt_imperial_olympic_institution_effect = yes
		trigger_event = kns_imperial_olympic.0032
	}
	else = {
		trigger_event = kns_imperial_olympic.0003
	}
}

kns_handle_imperial_olympic_organizer_death_effect = {
	if = {
		limit = { kns_has_olympic_settings_to_save_trigger = yes }
		kns_olympic_debug_log_effect = { MSG = organizer_death }
		kns_merge_character_hof_index_into_global_effect = yes
		kns_save_olympic_dormant_from_this_effect = yes
		if = {
			limit = { exists = player_heir }
			player_heir = {
				save_scope_as = kns_olympic_successor
			}
		}
		else_if = {
			limit = { exists = primary_heir }
			primary_heir = {
				save_scope_as = kns_olympic_successor
			}
		}
		if = {
			limit = { has_character_flag = kns_imperial_olympic_host }
			if = {
				limit = {
					is_ai = no
					exists = scope:kns_olympic_successor
					scope:kns_olympic_successor = { is_alive = yes }
				}
				kns_copy_olympic_runtime_to_effect = { TARGET = scope:kns_olympic_successor }
				kns_clear_imperial_olympic_vars_effect = yes
				scope:kns_olympic_successor = {
					save_scope_as = kns_emperor
					if = {
						limit = { has_variable = kns_olympic_bout_open }
						trigger_event = kns_imperial_olympic.0102
					}
					else = {
						trigger_event = {
							id = kns_imperial_olympic.0100
							days = 1
						}
					}
					trigger_event = kns_imperial_olympic.0031
				}
			}
			else = {
				kns_olympic_force_complete_bracket_effect = yes
			}
		}
	}
}

kns_add_imperial_olympic_entrant_effect = {
	$ENTRANT$ = {
		if = {
			limit = {
				scope:kns_emperor = {
					NOT = {
						is_target_in_variable_list = {
							name = kns_olympic_entrants
							target = $ENTRANT$
						}
					}
				}
			}
			if = {
				limit = { is_ai = no }
				add_character_flag = {
					flag = kns_entered_imperial_olympic
					days = 180
				}
			}

kns_gather_imperial_olympic_entrants_effect = {
	save_scope_as = kns_emperor
	clear_variable_list = kns_olympic_entrants
	clear_variable_list = kns_olympic_invited_rulers
	set_variable = {
		name = kns_olympic_invited_ruler_count
		value = 0
	}
	every_ruler = {
		limit = { kns_can_be_invited_imperial_olympic_ruler_trigger = yes }
		scope:kns_emperor = {
			change_variable = {
				name = kns_olympic_invited_ruler_count
				add = 1
			}
			add_to_variable_list = {
				name = kns_olympic_invited_rulers
				target = prev
			}
		}
	}
	# Reserve the player's chosen champions first so they cannot be crowded out.
	if = {
		limit = {
			variable_list_size = {
				name = kns_olympic_player_champion_picks
				value > 0
			}
		}
		every_in_list = {
			variable = kns_olympic_player_champion_picks
			limit = {
				is_alive = yes
				kns_can_enter_imperial_olympic_duel_trigger = yes
			}
			save_scope_as = kns_olympic_player_pick
			kns_add_imperial_olympic_entrant_effect = { ENTRANT = scope:kns_olympic_player_pick }
		}
	}
	kns_auto_invite_imperial_olympic_stars_effect = yes
	every_in_list = {
		variable = kns_olympic_invited_rulers
		save_scope_as = kns_olympic_invitee
		scope:kns_emperor = {
			set_variable = {
				name = kns_olympic_ruler_picks
				value = 0
			}
			set_variable = {
				name = kns_olympic_slots_per_ruler
				value = scope:kns_olympic_invitee.kns_olympic_slots_per_ruler_value
			}
			if = {
				limit = {
					scope:kns_olympic_invitee = { kns_can_enter_imperial_olympic_champion_trigger = yes }
					variable_list_size = {
						name = kns_olympic_entrants
						value < var:kns_olympic_bracket_size
					}
					var:kns_olympic_ruler_picks < var:kns_olympic_slots_per_ruler
				}
				kns_add_imperial_olympic_entrant_effect = { ENTRANT = scope:kns_olympic_invitee }
				change_variable = {
					name = kns_olympic_ruler_picks
					add = 1
				}
			}
			while = {
				count = 5
				limit = {
					var:kns_olympic_ruler_picks < var:kns_olympic_slots_per_ruler
					variable_list_size = {
						name = kns_olympic_entrants
						value < var:kns_olympic_bracket_size
					}
					scope:kns_olympic_invitee = {
						any_courtier = {
							kns_can_enter_imperial_olympic_champion_trigger = yes
							NOT = {
								scope:kns_emperor = {
									is_target_in_variable_list = {
										name = kns_olympic_entrants
										target = prev
									}
								}
							}
						}
					}
				}
				change_variable = {
					name = kns_olympic_ruler_picks
					add = 1
				}
				scope:kns_olympic_invitee = {
					random_courtier = {
						limit = {
							kns_can_enter_imperial_olympic_champion_trigger = yes
							NOT = {
								scope:kns_emperor = {
									is_target_in_variable_list = {
										name = kns_olympic_entrants
										target = prev
									}
								}
							}
						}
						weight = {
							base = 1
							modifier = {
								add = prowess_no_portrait
							}
						}
						save_scope_as = kns_olympic_champion_pick
					}
				}
				if = {
					limit = { exists = scope:kns_olympic_champion_pick }
					kns_add_imperial_olympic_entrant_effect = { ENTRANT = scope:kns_olympic_champion_pick }
				}
			}
			remove_variable = kns_olympic_ruler_picks
		}
	}
}

kns_seed_imperial_olympic_bracket_effect = {
	save_scope_as = kns_emperor
	clear_variable_list = kns_olympic_round_fighters
	clear_variable_list = kns_olympic_next_round
	set_variable = {
		name = kns_olympic_round
		value = 1
	}
	every_in_list = {
		variable = kns_olympic_player_champion_picks
		limit = {
			is_alive = yes
			kns_can_enter_imperial_olympic_duel_trigger = yes
			scope:kns_emperor = {
				is_target_in_variable_list = {
					name = kns_olympic_entrants
					target = prev
				}
				variable_list_size = {
					name = kns_olympic_round_fighters
					value < var:kns_olympic_bracket_size
				}
			}
		}
		save_scope_as = kns_olympic_seed_pick
		scope:kns_emperor = {
			remove_list_variable = {
				name = kns_olympic_entrants
				target = scope:kns_olympic_seed_pick
			}
			add_to_variable_list = {
				name = kns_olympic_round_fighters
				target = scope:kns_olympic_seed_pick
			}
		}
	}
	while = {
		count = 256
		limit = {
			variable_list_size = {
				name = kns_olympic_entrants
				value > 0
			}
			variable_list_size = {
				name = kns_olympic_round_fighters
				value < var:kns_olympic_bracket_size
			}
		}
		set_variable = {
			name = kns_olympic_found_seed
			value = 0
		}
		random_in_list = {
			variable = kns_olympic_entrants
			limit = {
				is_alive = yes
				kns_can_enter_imperial_olympic_duel_trigger = yes
			}
			weight = {
				base = 1
				modifier = {
					add = prowess_no_portrait
				}
			}
			save_scope_as = kns_olympic_seed_pick
			scope:kns_emperor = {
				set_variable = {
					name = kns_olympic_found_seed
					value = 1
				}
			}
		}
		if = {
			limit = {
				var:kns_olympic_found_seed = 1
				exists = scope:kns_olympic_seed_pick
			}
			remove_list_variable = {
				name = kns_olympic_entrants
				target = scope:kns_olympic_seed_pick
			}
			add_to_variable_list = {
				name = kns_olympic_round_fighters
				target = scope:kns_olympic_seed_pick
			}
		}
		else = {
			clear_variable_list = kns_olympic_entrants
		}
	}
}

			}

				name = kns_olympic_fighter_a

			}
			set_variable = {
				name = kns_olympic_fighter_b

			}

				limit = {
					OR = {

				}

kns_imperial_olympic_end_round_effect = {
	kns_olympic_save_host_effect = yes
	scope:kns_emperor = {
		if = {
			limit = {
				variable_list_size = {
					name = kns_olympic_next_round
					value = 1
				}
			}
			random_in_list = {
				variable = kns_olympic_next_round
				limit = { is_alive = yes }
				save_scope_as = kns_olympic_winner
			}
			if = {
				limit = { exists = scope:kns_olympic_winner }
				kns_olympic_debug_log_effect = { MSG = end_round_crown }
				kns_crown_imperial_olympic_champion_effect = { WINNER = scope:kns_olympic_winner }
			}
			else = {
				kns_olympic_debug_log_effect = { MSG = end_round_crown_failed }
				trigger_event = kns_imperial_olympic.0005
				kns_clear_imperial_olympic_vars_effect = yes
			}
		}
		else_if = {
			limit = {
				variable_list_size = {
					name = kns_olympic_next_round
					value = 0
				}
			}
			kns_olympic_debug_log_effect = { MSG = end_round_empty_next }
			trigger_event = kns_imperial_olympic.0005
			kns_clear_imperial_olympic_vars_effect = yes
		}
		else = {
			kns_olympic_debug_log_effect = { MSG = end_round_advance }
			every_in_list = {
				variable = kns_olympic_next_round
				limit = { is_alive = yes }
				save_scope_as = kns_olympic_advancer
				scope:kns_emperor = {
					add_to_variable_list = {
						name = kns_olympic_round_fighters
						target = scope:kns_olympic_advancer
					}
				}
			}
			clear_variable_list = kns_olympic_next_round
			change_variable = {
				name = kns_olympic_round
				add = 1
			}
			if = {
				limit = {
					variable_list_size = {
						name = kns_olympic_round_fighters
						value = 0
					}
				}
				trigger_event = kns_imperial_olympic.0005
				kns_clear_imperial_olympic_vars_effect = yes
			}
			else = {
				trigger_event = {
					id = kns_imperial_olympic.0100
					days = 1
				}
			}
		}
	}
}

kns_award_imperial_olympic_winner_effect = {
	$WINNER$ = {
		save_scope_as = kns_olympic_winner
		add_prestige = 5000
		create_character_memory = {
			type = kns_olympic_won_the_games
		}
		if = {
			limit = { NOT = { has_variable = kns_imperial_titles_won } }
			set_variable = {
				name = kns_imperial_titles_won
				value = 0
			}
		}
		change_variable = {
			name = kns_imperial_titles_won
			add = 1
		}
		kns_record_olympic_tournament_year_effect = yes
	}
	if = {
		limit = {
			scope:kns_emperor = {
				has_variable = kns_olympic_prize_gold
				var:kns_olympic_prize_gold >= 1000
			}
		}
		scope:kns_emperor = {
			add_gold = {
				value = var:kns_olympic_prize_gold
				multiply = -1
			}
		}
		$WINNER$ = {
			add_gold = scope:kns_emperor.kns_olympic_winner_gold_value
			if = {
				limit = { exists = dynasty }
				add_dynasty_prestige = scope:kns_emperor.kns_olympic_winner_renown_value
			}
		}
	}
	if = {
		limit = {
			NOT = { $WINNER$ = scope:kns_emperor }
			$WINNER$ = { kns_is_knight_trigger = no }
		}
		kns_become_a_knight_effect = {
			KNIGHT_TO_BE = $WINNER$
			KNIGHT_MAKER = scope:kns_emperor
		}
	}
	scope:kns_emperor = {
		kns_apply_olympic_knight_culture_modifier_effect = yes
		kns_apply_olympic_host_county_bonus_effect = yes
	}
	kns_register_imperial_hof_character_effect = { CHAR = $WINNER$ }
	$WINNER$ = {
		add_opinion = {
			target = scope:kns_emperor
			modifier = kns_olympic_champion_opinion
		}
	}
}

kns_crown_imperial_olympic_champion_effect = {
	$WINNER$ = {
		save_scope_as = kns_olympic_winner
	}
	kns_award_imperial_olympic_winner_effect = { WINNER = scope:kns_olympic_winner }
	if = {
		limit = {
			exists = scope:kns_emperor
			scope:kns_emperor = {
				is_alive = yes
				is_ai = no
			}
		}
		scope:kns_emperor = {
			trigger_event = kns_imperial_olympic.0010
		}
	}
	else_if = {
		limit = {
			any_player = {
				is_alive = yes
				OR = {
					has_character_flag = kns_olympic_institution
					highest_held_title_tier >= tier_empire
				}
			}
		}
		random_player = {
			limit = {
				is_alive = yes
				OR = {
					has_character_flag = kns_olympic_institution
					highest_held_title_tier >= tier_empire
				}
			}
			trigger_event = kns_imperial_olympic.0012
		}
	}
	else = {
		if = {
			limit = { exists = scope:kns_emperor }
			scope:kns_emperor = {
				kns_clear_imperial_olympic_vars_effect = yes
			}
		}
	}
}
