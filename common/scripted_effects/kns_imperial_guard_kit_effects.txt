# Imperial Guard kit is one bound weapon + plate per post.
# kns_ig_kit_kind: 1 weapon, 2 armor. kns_ig_kit_post: 1 axe, 2 hammer, 3 sword, 4 spear, 5 mace.

kns_ig_quiet_set_artifact_owner_effect = {
	set_variable = { name = kns_ig_quiet_move value = yes }
	set_owner = $OWNER$
}

kns_ig_stamp_kit_post_from_guard_effect = {
	if = {
		limit = { $GUARD$ = { has_character_flag = kns_ig_post_axe } }
		set_variable = { name = kns_ig_kit_post value = 1 }
	}
	else_if = {
		limit = { $GUARD$ = { has_character_flag = kns_ig_post_hammer } }
		set_variable = { name = kns_ig_kit_post value = 2 }
	}
	else_if = {
		limit = { $GUARD$ = { has_character_flag = kns_ig_post_sword } }
		set_variable = { name = kns_ig_kit_post value = 3 }
	}
	else_if = {
		limit = { $GUARD$ = { has_character_flag = kns_ig_post_spear } }
		set_variable = { name = kns_ig_kit_post value = 4 }
	}
	else_if = {
		limit = { $GUARD$ = { has_character_flag = kns_ig_post_mace } }
		set_variable = { name = kns_ig_kit_post value = 5 }
	}
}

kns_ig_stamp_kit_artifact_effect = {
	set_variable = { name = kns_ig_kit_kind value = $KIND$ }
	set_variable = {
		name = kns_ig_kit_guard
		value = $GUARD$
	}
	set_variable = {
		name = kns_ig_kit_emperor
		value = $EMPEROR$
	}
	kns_ig_stamp_kit_post_from_guard_effect = { GUARD = $GUARD$ }
}

		subtract = artifact_durability
	}
}

kns_ig_relink_or_queue_duplicate_kit_slot_effect = {
	if = {
		limit = {
			var:kns_ig_kit_kind = $KIND$
			var:kns_ig_kit_post = $POST$
		}
		if = {
			limit = {
				$EMPEROR$ = { exists = var:kns_ig_post_$SLOT$ }
				NOT = { this = $EMPEROR$.var:kns_ig_post_$SLOT$ }
			}
			add_to_list = kns_ig_dup_kit
		}
		else_if = {
			limit = { $EMPEROR$ = { NOT = { exists = var:kns_ig_post_$SLOT$ } } }
			$EMPEROR$ = {
				set_variable = {
					name = kns_ig_post_$SLOT$
					value = prev
				}
			}
		}
	}
}

kns_ig_start_kit_recovery_effect = {
	if = {
		limit = {
			$GUARD$ = { has_character_flag = kns_ig_kit_recovery_lock }
		}
	}
	else_if = {
		limit = {
			$GUARD$ = {
				is_alive = yes
				kns_is_imperial_guard_trigger = yes
			}
			$HOLDER$ = {
				is_alive = yes
				NOT = { this = $GUARD$ }
				kns_ig_kit_holder_gives_back_quietly_trigger = { GUARD = $GUARD$ }
			}
		}
		kns_ig_return_stolen_kit_to_guard_effect = {
			GUARD = $GUARD$
			HOLDER = $HOLDER$
		}
		$GUARD$ = { kns_ig_give_accolade_title_nickname_effect = yes }
	}
	else_if = {
		limit = {
			$GUARD$ = {
				is_alive = yes
				kns_is_imperial_guard_trigger = yes
				NOT = { has_trait = incapable }
				NOT = { has_variable = engaged_in_single_combat }
			}
			$HOLDER$ = {
				is_alive = yes
				is_ai = yes
				is_imprisoned = no
				NOT = { has_trait = incapable }
				NOT = { this = $GUARD$ }
			}
		}
		$GUARD$ = { add_character_flag = kns_ig_kit_recovery_lock }
		$GUARD$ = { save_scope_as = kns_ig_guard }
		$HOLDER$ = { save_scope_as = kns_ig_holder }
		if = {
			limit = { $GUARD$ = { exists = var:kns_imperial_guard_liege } }
			$GUARD$.var:kns_imperial_guard_liege = { save_scope_as = kns_ig_emperor }
		}
		else_if = {
			limit = { exists = var:kns_ig_kit_emperor }
			var:kns_ig_kit_emperor = { save_scope_as = kns_ig_emperor }
		}
		if = {
			limit = {
				$HOLDER$ = {
					kns_ig_kit_holder_returns_trigger = { GUARD = $GUARD$ }
				}
			}
			if = {
				limit = { exists = scope:kns_ig_emperor }
				scope:kns_ig_emperor = { trigger_event = kns_ig_kit.0001 }
			}
			else = {
				$GUARD$ = { trigger_event = kns_ig_kit.0001 }
			}
		}
		else = {
			if = {
				limit = { exists = scope:kns_ig_emperor }
				scope:kns_ig_emperor = { trigger_event = kns_ig_kit.0002 }
			}
			else = {
				$GUARD$ = { trigger_event = kns_ig_kit.0002 }
			}
		}
	}
}

kns_ig_try_recover_if_kit_stolen_effect = {
	# this = artifact
	if = {
		limit = { exists = var:kns_ig_quiet_move }
		remove_variable = kns_ig_quiet_move
	}
	else_if = {
		limit = {
			exists = var:kns_ig_kit_kind
			exists = var:kns_ig_kit_guard
			exists = scope:owner
			var:kns_ig_kit_guard = {
				is_alive = yes
				kns_is_imperial_guard_trigger = yes
			}
			NOT = { scope:owner = var:kns_ig_kit_guard }
			trigger_if = {
				limit = { exists = var:kns_ig_kit_emperor }
				NOT = { scope:owner = var:kns_ig_kit_emperor }
			}
			trigger_else = {
				always = yes
			}
		}
		kns_ig_start_kit_recovery_effect = {
			GUARD = var:kns_ig_kit_guard
			HOLDER = scope:owner
		}
	}
}

kns_ig_check_guard_kit_in_wrong_hands_effect = {
	# this = guard
	if = {
		limit = {
			exists = var:kns_ig_kit_weapon
			NOT = { has_character_flag = kns_ig_kit_recovery_lock }
			var:kns_ig_kit_weapon = {
				exists = artifact_owner
				NOT = { artifact_owner = prev }
				trigger_if = {
					limit = { exists = var:kns_ig_kit_emperor }
					NOT = { artifact_owner = var:kns_ig_kit_emperor }
				}
			}
		}
		var:kns_ig_kit_weapon = {
			save_scope_as = kns_ig_stolen_piece
			artifact_owner = { save_scope_as = kns_ig_wrong_holder }
		}
		kns_ig_start_kit_recovery_effect = {
			GUARD = this
			HOLDER = scope:kns_ig_wrong_holder
		}
	}
	else_if = {
		limit = {
			exists = var:kns_ig_kit_armor
			NOT = { has_character_flag = kns_ig_kit_recovery_lock }
			var:kns_ig_kit_armor = {
				exists = artifact_owner
				NOT = { artifact_owner = prev }
				trigger_if = {
					limit = { exists = var:kns_ig_kit_emperor }
					NOT = { artifact_owner = var:kns_ig_kit_emperor }
				}
			}
		}
		var:kns_ig_kit_armor = {
			artifact_owner = { save_scope_as = kns_ig_wrong_holder }
		}
		kns_ig_start_kit_recovery_effect = {
			GUARD = this
			HOLDER = scope:kns_ig_wrong_holder
		}
	}
}
