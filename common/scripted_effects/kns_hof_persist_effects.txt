# Split Hall of Legends freeze pools (100 title slots, 100 duel slots).
# Living rank from kns_hof_index. Dead freeze into the matching pool only.

kns_hof_consider_dead_champion_slot_effect = {
	if = {
		limit = {
			has_global_variable = kns_hof_title_stat_$SLOT$
			global_var:kns_hof_title_stat_$SLOT$ > 0
			NOT = { exists = var:kns_hof_used_champ_$SLOT$ }
			global_var:kns_hof_title_stat_$SLOT$ > var:kns_hof_best_score
			OR = {
				NOT = { has_global_variable = kns_hof_title_ref_$SLOT$ }
				NOT = { exists = global_var:kns_hof_title_ref_$SLOT$ }
				global_var:kns_hof_title_ref_$SLOT$ = { is_alive = no }
			}
			NAND = {
				exists = global_var:kns_hof_title_ref_$SLOT$
				global_var:kns_hof_title_ref_$SLOT$ = { has_variable = kns_hof_champ_picked }
			}
		}
		set_variable = {
			name = kns_hof_best_score
			value = global_var:kns_hof_title_stat_$SLOT$
		}
		set_variable = {
			name = kns_hof_best_slot
			value = $SLOT$
		}
	}
}

kns_hof_consider_steal_title_slot_effect = {
	if = {
		limit = {
			OR = {
				has_global_variable = kns_hof_title_ref_$SLOT$
				has_global_variable = kns_hof_title_stat_$SLOT$
			}
			NAND = {
				exists = global_var:kns_hof_title_ref_$SLOT$
				global_var:kns_hof_title_ref_$SLOT$ = scope:kns_hof_char
			}
		}
		set_variable = {
			name = kns_hof_slot_score
			value = 0
		}
		if = {
			limit = { has_global_variable = kns_hof_title_stat_$SLOT$ }
			set_variable = {
				name = kns_hof_slot_score
				value = global_var:kns_hof_title_stat_$SLOT$
			}
		}
		if = {
			limit = {
				exists = global_var:kns_hof_title_ref_$SLOT$
				global_var:kns_hof_title_ref_$SLOT$ = { is_alive = yes }
				var:kns_hof_slot_score < var:kns_hof_steal_living_score
			}
			set_variable = {
				name = kns_hof_steal_living_score
				value = var:kns_hof_slot_score
			}
			set_variable = {
				name = kns_hof_steal_living_slot
				value = $SLOT$
			}
		}
		if = {
			limit = { var:kns_hof_slot_score < var:kns_hof_steal_any_score }
			set_variable = {
				name = kns_hof_steal_any_score
				value = var:kns_hof_slot_score
			}
			set_variable = {
				name = kns_hof_steal_any_slot
				value = $SLOT$
			}
		}
	}
}

			limit = { exists = global_var:kns_hof_title_ref_$SLOT$ }

				if = {
					limit = { has_variable = kns_hof_title_slot }

kns_hof_pick_steal_victim_effect = {
	remove_variable = kns_hof_steal_slot
	remove_variable = kns_hof_victim_score
	if = {
		limit = { exists = var:kns_hof_steal_living_slot }
		set_variable = {
			name = kns_hof_steal_slot
			value = var:kns_hof_steal_living_slot
		}
		set_variable = {
			name = kns_hof_victim_score
			value = var:kns_hof_steal_living_score
		}
	}
	else_if = {
		limit = { exists = var:kns_hof_steal_any_slot }
		set_variable = {
			name = kns_hof_steal_slot
			value = var:kns_hof_steal_any_slot
		}
		set_variable = {
			name = kns_hof_victim_score
			value = var:kns_hof_steal_any_score
		}
	}
}

kns_hof_migrate_old_shared_slot_effect = {
	if = {
		limit = {
			has_global_variable = kns_hof_stat_titles_$SLOT$
			global_var:kns_hof_stat_titles_$SLOT$ > 0
			OR = {
				NOT = { has_global_variable = kns_hof_ref_$SLOT$ }
				NOT = { exists = global_var:kns_hof_ref_$SLOT$ }
				global_var:kns_hof_ref_$SLOT$ = { is_alive = no }
			}
		}
		if = {
			limit = { exists = global_var:kns_hof_ref_$SLOT$ }
			global_var:kns_hof_ref_$SLOT$ = { save_scope_as = kns_hof_char }
			kns_hof_freeze_dead_title_effect = yes
		}
	}
	if = {
		limit = {
			has_global_variable = kns_hof_stat_duels_$SLOT$
			global_var:kns_hof_stat_duels_$SLOT$ > 0
			OR = {
				NOT = { has_global_variable = kns_hof_ref_$SLOT$ }
				NOT = { exists = global_var:kns_hof_ref_$SLOT$ }
				global_var:kns_hof_ref_$SLOT$ = { is_alive = no }
			}
		}
		if = {
			limit = { exists = global_var:kns_hof_ref_$SLOT$ }
			global_var:kns_hof_ref_$SLOT$ = { save_scope_as = kns_hof_char }
			kns_hof_freeze_dead_duel_effect = yes
		}
	}
}

kns_hof_pick_next_hybrid_champion_effect = {
	if = {
		limit = { exists = scope:kns_hof_best_living }
		clear_saved_scope = kns_hof_best_living
	}
	ordered_in_global_list = {
		variable = kns_hof_index
		limit = {
			NOT = { has_variable = kns_hof_champ_picked }
			has_variable = kns_imperial_titles_won
			var:kns_imperial_titles_won > 0
		}
		order_by = kns_hof_titles_rank_value
		max = 1
		check_range_bounds = no
		save_scope_as = kns_hof_best_living
	}
	scope:kns_hof_viewer = {
		set_variable = {
			name = kns_hof_living_score
			value = -1
		}
		if = {
			limit = { exists = scope:kns_hof_best_living }
			set_variable = {
				name = kns_hof_living_score
				value = scope:kns_hof_best_living.kns_hof_titles_rank_value
			}
		}
		set_variable = {
			name = kns_hof_best_score
			value = -1
		}
		remove_variable = kns_hof_best_slot
		kns_hof_scan_dead_champion_slots_effect = yes
		if = {
			limit = {
				exists = scope:kns_hof_best_living
				OR = {
					NOT = { exists = var:kns_hof_best_slot }
					var:kns_hof_living_score >= var:kns_hof_best_score
				}
			}
			scope:kns_hof_best_living = {
				save_scope_as = kns_hof_rank_char
				set_variable = kns_hof_champ_picked
			}
			kns_hof_copy_champion_from_char_to_tmp_effect = yes
			kns_hof_store_champion_display_effect = yes
			change_variable = {
				name = kns_hof_display_i
				add = 1
			}
		}
		else_if = {
			limit = { exists = var:kns_hof_best_slot }
			kns_hof_scan_load_title_tmp_effect = yes
			if = {
				limit = { exists = var:kns_hof_tmp_char }
				var:kns_hof_tmp_char = {
					set_variable = kns_hof_champ_picked
				}
			}
			kns_hof_store_champion_display_effect = yes
			kns_hof_mark_champion_used_effect = yes
			change_variable = {
				name = kns_hof_display_i
				add = 1
			}
		}
	}
}
