# After a normal decoration roll: force prowess to 11, mark as a
# Christian relic, and write relic history. Other modifiers stay
# whatever the vanilla roll gave.
kns_strip_artifact_prowess_modifiers_effect = {
	if = { limit = { has_artifact_modifier = artifact_prowess_1_modifier } remove_artifact_modifier = artifact_prowess_1_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_2_modifier } remove_artifact_modifier = artifact_prowess_2_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_3_modifier } remove_artifact_modifier = artifact_prowess_3_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_4_modifier } remove_artifact_modifier = artifact_prowess_4_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_5_modifier } remove_artifact_modifier = artifact_prowess_5_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_6_modifier } remove_artifact_modifier = artifact_prowess_6_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_7_modifier } remove_artifact_modifier = artifact_prowess_7_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_8_modifier } remove_artifact_modifier = artifact_prowess_8_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_9_modifier } remove_artifact_modifier = artifact_prowess_9_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_10_modifier } remove_artifact_modifier = artifact_prowess_10_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_11_modifier } remove_artifact_modifier = artifact_prowess_11_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_1_negative_modifier } remove_artifact_modifier = artifact_prowess_1_negative_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_2_negative_modifier } remove_artifact_modifier = artifact_prowess_2_negative_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_no_decorations_modifier } remove_artifact_modifier = artifact_prowess_no_decorations_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_no_decorations_medium_tier_modifier } remove_artifact_modifier = artifact_prowess_no_decorations_medium_tier_modifier }
	if = { limit = { has_artifact_modifier = artifact_prowess_no_decorations_high_tier_modifier } remove_artifact_modifier = artifact_prowess_no_decorations_high_tier_modifier }
}

kns_finalize_prodigious_artifact_effect = {
	kns_strip_artifact_prowess_modifiers_effect = yes
	add_artifact_modifier = artifact_prowess_11_modifier
	set_variable = {
		name = relic
		value = flag:christian
	}
	set_variable = {
		name = historical_unique_artifact
		value = yes
	}
	add_artifact_history = {
		type = created_before_history
	}
	if = {
		limit = {
			exists = scope:kns_kit_smith
			exists = scope:kns_kit_owner
		}
		add_artifact_history = {
			type = given
			actor = scope:kns_kit_smith
			recipient = scope:kns_kit_owner
		}
	}
}

kns_apply_vanilla_weapon_roll_effect = {
	add_scaled_artifact_modifier_prowess_effect = yes
	add_2_scaled_artifact_modifier_combat_effect = yes
}

kns_remember_prodigious_weapon_kind_effect = {
	if = {
		limit = { exists = scope:weapon_type scope:weapon_type = flag:artifact_weapon_type_sword }
		save_scope_value_as = { name = kns_prodigious_weapon_kind value = flag:sword }
		scope:kns_kit_owner = { set_variable = { name = kns_knight_weapon_type value = flag:sword } }
	}
	else_if = {
		limit = { exists = scope:weapon_type scope:weapon_type = flag:artifact_weapon_type_axe }
		save_scope_value_as = { name = kns_prodigious_weapon_kind value = flag:axe }
		scope:kns_kit_owner = { set_variable = { name = kns_knight_weapon_type value = flag:axe } }
	}
	else_if = {
		limit = { exists = scope:weapon_type scope:weapon_type = flag:artifact_weapon_type_hammer }
		save_scope_value_as = { name = kns_prodigious_weapon_kind value = flag:hammer }
		scope:kns_kit_owner = { set_variable = { name = kns_knight_weapon_type value = flag:hammer } }
	}
	else_if = {
		limit = { exists = scope:weapon_type scope:weapon_type = flag:artifact_weapon_type_mace }
		save_scope_value_as = { name = kns_prodigious_weapon_kind value = flag:mace }
		scope:kns_kit_owner = { set_variable = { name = kns_knight_weapon_type value = flag:mace } }
	}
	else = {
		save_scope_value_as = { name = kns_prodigious_weapon_kind value = flag:spear }
		scope:kns_kit_owner = { set_variable = { name = kns_knight_weapon_type value = flag:spear } }
	}
}

kns_create_prodigious_armor_effect = {
	set_artifact_rarity_illustrious = yes
	scope:kns_kit_owner = { save_scope_as = owner }
	scope:kns_kit_smith = { save_scope_as = smith }
	random_list = {
		1 = {
			create_artifact = {
				name = artifact_armor_name
				description = placeholder
				visuals = armor
				type = armor_mail
				modifier = artifact_prowess_1_modifier
				template = kns_prodigious_relic_template
				wealth = scope:wealth
				quality = scope:quality
				creator = scope:kns_kit_smith
				decaying = no
				save_scope_as = newly_created_artifact
			}
		}
		1 = {
			create_artifact = {
				name = artifact_armor_name
				description = placeholder
				visuals = armor
				type = armor_scale
				modifier = artifact_prowess_1_modifier
				template = kns_prodigious_relic_template
				wealth = scope:wealth
				quality = scope:quality
				creator = scope:kns_kit_smith
				decaying = no
				save_scope_as = newly_created_artifact
			}
		}
		1 = {
			create_artifact = {
				name = artifact_armor_name
				description = placeholder
				visuals = armor
				type = armor_plate
				modifier = artifact_prowess_1_modifier
				template = kns_prodigious_relic_template
				wealth = scope:wealth
				quality = scope:quality
				creator = scope:kns_kit_smith
				decaying = no
				save_scope_as = newly_created_artifact
			}
		}
	}
	scope:newly_created_artifact ?= {
		set_artifact_feature_group = armor_decoration
		set_artifact_feature_group = armor_property
		get_artifact_feature_references_effect = yes
		add_scaled_artifact_modifier_prowess_effect = yes
		add_scaled_artifact_modifier_combat_effect = yes
		kns_finalize_prodigious_artifact_effect = yes
		kns_name_prodigious_armor_effect = yes
		if = {
			limit = { scope:kns_kit_owner = { has_character_flag = kns_papal_tourney_prize } }
			set_artifact_description = kns_art_papal_tourney_armor_desc
		}
		if = {
			limit = {
				scope:kns_kit_owner = { can_equip_artifact = prev }
			}
			equip_artifact_to_owner = yes
		}
		else = {
			equip_artifact_to_owner_replace = yes
		}
	}
}
