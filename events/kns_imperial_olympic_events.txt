# Continue a predecessor's tradition
kns_imperial_olympic.0030 = {
	type = character_event
	title = kns_imperial_olympic.0030.t
	desc = kns_imperial_olympic.0030.desc
	theme = martial
	override_background = { reference = courtyard }

	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		trigger = { exists = scope:kns_olympic_predecessor }
		character = scope:kns_olympic_predecessor
		animation = personality_honorable
	}

	immediate = {
		add_character_flag = kns_olympic_continue_session
		kns_load_olympic_dormant_onto_this_effect = yes
		if = {
			limit = { exists = global_var:kns_olympic_dormant_predecessor }
			global_var:kns_olympic_dormant_predecessor = {
				save_scope_as = kns_olympic_predecessor
				kns_merge_character_hof_index_into_global_effect = yes
			}
		}
	}

	option = {
		name = kns_imperial_olympic.0030.a
		custom_tooltip = kns_imperial_olympic.0030.a.tt
		trigger_event = kns_imperial_olympic.0002
	}

	option = {
		name = kns_imperial_olympic.0030.b
		custom_tooltip = kns_imperial_olympic.0030.b.tt
		hidden_effect = {
			remove_character_flag = kns_olympic_continue_session
			kns_clear_olympic_dormant_effect = yes
			if = {
				limit = { NOT = { has_character_flag = kns_imperial_olympic_host } }
				remove_variable = kns_olympic_interval_years
				remove_variable = kns_olympic_prize_gold
				remove_variable = kns_olympic_bracket_size
				remove_variable = kns_olympic_location
				remove_variable = kns_olympic_next_year
			}
		}
	}
}

kns_imperial_olympic.0010 = {
	type = character_event
	title = kns_imperial_olympic.0010.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:kns_olympic_winner
					this = scope:kns_olympic_winner
				}
				desc = kns_imperial_olympic.0010.desc_self
			}
			triggered_desc = {
				trigger = { exists = scope:kns_olympic_winner }
				desc = kns_imperial_olympic.0010.desc_crown
			}
			desc = kns_imperial_olympic.0010.desc_none
		}
	}
	theme = martial
	override_background = { reference = courtyard }

	left_portrait = {
		character = root
		animation = personality_honorable
	}
	right_portrait = {
		trigger = {
			exists = scope:kns_olympic_winner
			NOT = { this = scope:kns_olympic_winner }
		}
		character = scope:kns_olympic_winner
		animation = happiness
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:kns_emperor } }
			save_scope_as = kns_emperor
		}
	}

	option = {
		name = kns_imperial_olympic.0010.a
		trigger = {
			exists = scope:kns_olympic_winner
			this = scope:kns_olympic_winner
		}
		custom_tooltip = kns_imperial_olympic_prize_tt
		hidden_effect = {
			kns_clear_imperial_olympic_vars_effect = yes
		}
	}

	option = {
		name = kns_imperial_olympic.0010.b
		trigger = {
			exists = scope:kns_olympic_winner
			NOT = { this = scope:kns_olympic_winner }
		}
		custom_tooltip = kns_imperial_olympic_prize_tt
		hidden_effect = {
			if = {
				limit = {
					scope:kns_olympic_winner = { is_ai = no }
				}
				scope:kns_olympic_winner = {
					trigger_event = kns_imperial_olympic.0011
				}
			}
			else = {
				kns_clear_imperial_olympic_vars_effect = yes
			}
		}
	}

	option = {
		name = kns_imperial_olympic.0010.c
		trigger = {
			NOT = { exists = scope:kns_olympic_winner }
		}
		hidden_effect = {
			kns_clear_imperial_olympic_vars_effect = yes
		}
	}
}

# Hidden prowess duel (AI vs AI)
kns_imperial_olympic.0101 = {
	type = character_event
	hidden = yes

	immediate = {
		kns_restore_imperial_olympic_bout_scopes_effect = yes
		if = {
			limit = { has_variable = kns_olympic_bout_open }
			random_list = {
				50 = {
					modifier = {
						add = scope:kns_olympic_fighter_a.prowess_no_portrait
					}
					scope:kns_olympic_fighter_a = {
						save_scope_as = kns_olympic_winner
					}
				}
				50 = {
					modifier = {
						add = scope:kns_olympic_fighter_b.prowess_no_portrait
					}
					scope:kns_olympic_fighter_b = {
						save_scope_as = kns_olympic_winner
					}
				}
			}
			if = {
				limit = { exists = scope:kns_olympic_winner }
				kns_record_imperial_olympic_bout_win_from_pair_effect = {
					WINNER = scope:kns_olympic_winner
				}
			}
			else = {
				scope:kns_olympic_fighter_a = {
					save_scope_as = kns_olympic_winner
				}
				kns_record_imperial_olympic_bout_win_from_pair_effect = {
					WINNER = scope:kns_olympic_winner
				}
			}
		}
	}
}

# Single combat outcome
kns_imperial_olympic.0103 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { exists = scope:sc_victor }
			scope:sc_victor = {
				save_scope_as = kns_olympic_winner
			}
		}
		kns_restore_imperial_olympic_bout_scopes_effect = yes
		kns_olympic_debug_log_effect = { MSG = player_bout_0103 }
		scope:kns_emperor = {
			if = {
				limit = { has_variable = kns_olympic_bout_open }
				if = {
					limit = {
						exists = scope:sc_attacker
						exists = scope:sc_defender
					}
					single_combat_clean_shirtlessness_effect = {
						ATTACKER = scope:sc_attacker
						DEFENDER = scope:sc_defender
					}
				}
				if = {
					limit = { NOT = { exists = scope:kns_olympic_winner } }
					if = {
						limit = {
							exists = scope:kns_olympic_fighter_a
							exists = scope:kns_olympic_fighter_b
							scope:kns_olympic_fighter_a = {
								prowess_no_portrait >= scope:kns_olympic_fighter_b.prowess_no_portrait
							}
						}
						scope:kns_olympic_fighter_a = {
							save_scope_as = kns_olympic_winner
						}
					}
					else_if = {
						limit = { exists = scope:kns_olympic_fighter_b }
						scope:kns_olympic_fighter_b = {
							save_scope_as = kns_olympic_winner
						}
					}
					else_if = {
						limit = { exists = scope:kns_olympic_fighter_a }
						scope:kns_olympic_fighter_a = {
							save_scope_as = kns_olympic_winner
						}
					}
				}
				if = {
					limit = { exists = scope:kns_olympic_winner }
					kns_record_imperial_olympic_bout_win_from_pair_effect = {
						WINNER = scope:kns_olympic_winner
					}
				}
				else = {
					kns_olympic_debug_log_effect = { MSG = player_bout_0103_no_winner }
					remove_variable = kns_olympic_bout_open
					trigger_event = {
						id = kns_imperial_olympic.0100
						days = 1
					}
				}
			}
			else = {
				kns_olympic_debug_log_effect = { MSG = player_bout_0103_ignored_no_open }
			}
		}
	}
}
