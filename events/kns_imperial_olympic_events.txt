# Continue a predecessor's tradition
kns_imperial_olympic.0030 = {
	type = character_event
	title = kns_imperial_olympic.0030.t
	desc = kns_imperial_olympic.0030.desc
	theme = martial
	override_background = { reference = courtyard }

	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		trigger = { exists = scope:kns_olympic_predecessor }
		character = scope:kns_olympic_predecessor
		animation = personality_honorable
	}

	immediate = {
		add_character_flag = kns_olympic_continue_session
		kns_load_olympic_dormant_onto_this_effect = yes
		if = {
			limit = { exists = global_var:kns_olympic_dormant_predecessor }
			global_var:kns_olympic_dormant_predecessor = {
				save_scope_as = kns_olympic_predecessor
				kns_merge_character_hof_index_into_global_effect = yes
			}
		}
	}

	option = {
		name = kns_imperial_olympic.0030.a
		custom_tooltip = kns_imperial_olympic.0030.a.tt
		trigger_event = kns_imperial_olympic.0002
	}

	option = {
		name = kns_imperial_olympic.0030.b
		custom_tooltip = kns_imperial_olympic.0030.b.tt
		hidden_effect = {
			remove_character_flag = kns_olympic_continue_session
			kns_clear_olympic_dormant_effect = yes
			if = {
				limit = { NOT = { has_character_flag = kns_imperial_olympic_host } }
				remove_variable = kns_olympic_interval_years
				remove_variable = kns_olympic_prize_gold
				remove_variable = kns_olympic_bracket_size
				remove_variable = kns_olympic_location
				remove_variable = kns_olympic_next_year
			}
		}
	}
}

# Emperor crowns the bracket champion
kns_imperial_olympic.0010 = {
	type = character_event
	title = kns_imperial_olympic.0010.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:kns_olympic_winner
					this = scope:kns_olympic_winner
				}
				desc = kns_imperial_olympic.0010.desc_self
			}
			triggered_desc = {
				trigger = { exists = scope:kns_olympic_winner }
				desc = kns_imperial_olympic.0010.desc_crown
			}
			desc = kns_imperial_olympic.0010.desc_none
		}
	}
	theme = martial
	override_background = { reference = courtyard }

	left_portrait = {
		character = scope:kns_olympic_winner
		scripted_animation = duel_celebrate
	}
	right_portrait = {
		trigger = {
			exists = scope:kns_olympic_winner
			NOT = { this = scope:kns_olympic_winner }
		}
		character = root
		animation = personality_honorable
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:kns_emperor } }
			save_scope_as = kns_emperor
		}
		kns_olympic_restore_champion_scope_effect = yes
	}

	option = {
		name = kns_imperial_olympic.0010.a
		trigger = {
			exists = scope:kns_olympic_winner
			this = scope:kns_olympic_winner
		}
		custom_tooltip = kns_imperial_olympic_prize_tt
		hidden_effect = {
			kns_olympic_clear_champion_scope_effect = yes
			kns_clear_imperial_olympic_vars_effect = yes
		}
	}

	option = {
		name = kns_imperial_olympic.0010.b
		trigger = {
			exists = scope:kns_olympic_winner
			NOT = { this = scope:kns_olympic_winner }
		}
		custom_tooltip = kns_imperial_olympic_prize_tt
		hidden_effect = {
			if = {
				limit = {
					scope:kns_olympic_winner = { is_ai = no }
				}
				scope:kns_olympic_winner = {
					trigger_event = kns_imperial_olympic.0009
				}
			}
			else = {
				kns_olympic_clear_champion_scope_effect = yes
				kns_clear_imperial_olympic_vars_effect = yes
			}
		}
	}

	option = {
		name = kns_imperial_olympic.0010.c
		trigger = {
			NOT = { exists = scope:kns_olympic_winner }
		}
		hidden_effect = {
			kns_olympic_clear_champion_scope_effect = yes
			kns_clear_imperial_olympic_vars_effect = yes
		}
	}
}

# Single combat outcome. Both combatants must be the stored pair; player-as-victor alone is not enough.
kns_imperial_olympic.0103 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { exists = scope:sc_victor }
			scope:sc_victor = {
				save_scope_as = kns_olympic_combat_victor
				save_scope_as = kns_olympic_winner
			}
		}
		if = {
			limit = { exists = scope:sc_loser }
			scope:sc_loser = {
				save_scope_as = kns_olympic_combat_loser
			}
		}
		if = {
			limit = { exists = scope:sc_attacker }
			scope:sc_attacker = {
				save_scope_as = kns_olympic_combat_attacker
			}
		}
		if = {
			limit = { exists = scope:sc_defender }
			scope:sc_defender = {
				save_scope_as = kns_olympic_combat_defender
			}
		}
		kns_restore_imperial_olympic_bout_scopes_effect = yes
		kns_olympic_debug_log_bout_effect = { MSG = player_bout_0103 }
		scope:kns_emperor = {
			if = {
				limit = { has_variable = kns_olympic_bout_open }
				if = {
					limit = {
						exists = scope:kns_olympic_combat_attacker
						exists = scope:kns_olympic_combat_defender
					}
					single_combat_clean_shirtlessness_effect = {
						ATTACKER = scope:kns_olympic_combat_attacker
						DEFENDER = scope:kns_olympic_combat_defender
					}
				}
				if = {
					limit = { kns_olympic_combat_matches_open_bout_trigger = yes }
					kns_olympic_debug_log_bout_effect = { MSG = player_bout_0103_match_ok }
					kns_olympic_apply_minigame_result_effect = yes
					kns_olympic_advance_after_player_round_effect = yes
				}
				else = {
					kns_olympic_debug_log_bout_effect = { MSG = player_bout_0103_ignored_mismatch }
				}
			}
			else = {
				kns_olympic_debug_log_bout_effect = { MSG = player_bout_0103_ignored_no_open }
			}
		}
		kns_olympic_reset_character_duel_state_effect = { CHAR = scope:kns_olympic_player_combatant }
		kns_olympic_reset_character_duel_state_effect = { CHAR = scope:kns_olympic_ai_combatant }
		kns_olympic_reset_character_duel_state_effect = { CHAR = scope:kns_olympic_fighter_a }
		kns_olympic_reset_character_duel_state_effect = { CHAR = scope:kns_olympic_fighter_b }
		kns_olympic_reset_character_duel_state_effect = { CHAR = scope:sc_loser }
		kns_olympic_reset_character_duel_state_effect = { CHAR = scope:sc_victor }
		kns_olympic_seed_fresh_duel_variables_effect = { CHAR = scope:kns_olympic_player_combatant }
		kns_olympic_seed_fresh_duel_variables_effect = { CHAR = scope:kns_olympic_ai_combatant }
		kns_olympic_seed_fresh_duel_variables_effect = { CHAR = scope:kns_olympic_fighter_a }
		kns_olympic_seed_fresh_duel_variables_effect = { CHAR = scope:kns_olympic_fighter_b }
		kns_olympic_clear_stale_combat_scopes_effect = yes
	}
}
