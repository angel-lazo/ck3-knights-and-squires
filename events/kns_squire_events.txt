namespace = kns_squire

scripted_trigger kns_available_for_squire_event = {
	is_available = yes
	kns_is_squire_with_knight = yes
	NOT = { has_character_flag = kns_squire_event_$NUM$ }
}

# Horsemanship
kns_squire.0001 = {
	type = character_event
	title = kns_squire.0001.t
	desc = kns_squire.0001.desc
	theme = martial
	override_background = { reference = army_camp }

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = scope:my_knight
		animation = personality_honorable
	}

	trigger = {
		kns_available_for_squire_event = { NUM = 0001 }
	}

	immediate = {
		add_character_flag = { flag = kns_squire_event_0001 months = 4 }
		random_relation = {
			type = kns_knight
			save_scope_as = my_knight
		}
	}

	option = {
		name = kns_squire.0001.a
		duel = {
			skill = martial
			value = medium_skill_rating
			55 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -39
				}
				desc = kns_squire.tt.success
				send_interface_message = {
					type = event_toast_effect_good
					title = kns_squire.tt.success
					desc = kns_squire.desc.success
					kns_add_squire_trait_xp_effect = { MIN_VALUE = 4 MAX_VALUE = 6 }
					kns_try_squire_skill_increase_effect = { SKILL = martial }
					left_icon = root
					right_icon = scope:my_knight
				}
			}
			45 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -19
				}
				desc = kns_squire.tt.failure
				send_interface_message = {
					type = event_toast_effect_bad
					title = kns_squire.tt.failure
					desc = kns_squire.desc.failure
					kns_add_squire_trait_xp_effect = { MIN_VALUE = -4 MAX_VALUE = -1 }
					left_icon = root
					right_icon = scope:my_knight
				}
			}
		}
	}
}

# Directed training from the knight
kns_squire.0400 = {
	type = character_event
	title = kns_squire.0400.t
	desc = kns_squire.0400.desc
	theme = martial
	override_background = { reference = courtyard }

	left_portrait = {
		character = root
		animation = idle
	}
	right_portrait = {
		character = scope:my_squire
		animation = idle
	}

	# Strategy
	option = {
		name = kns_squire.0400.a
		random_list = {
			3 = {
				modifier = {
					scope:my_squire.martial > mediocre_skill_rating
					add = 2
				}
				modifier = {
					scope:my_squire = { has_trait = shrewd }
					add = 2
				}
				custom_tooltip = kns_squire.0400.good_training.tt
				scope:my_squire = {
					kns_add_squire_trait_xp_effect = { MIN_VALUE = 8 MAX_VALUE = 14 }
					kns_try_squire_skill_increase_effect = { SKILL = martial }
					send_interface_message = {
						type = event_toast_effect_good
						title = kns_squire.0400.had_a_training_session.t
						desc = kns_squire.0401_good_strategy.desc
						left_icon = scope:my_squire
						right_icon = root
					}
				}
			}
			1 = {
				modifier = {
					scope:my_squire.martial < mediocre_skill_rating
					add = 2
				}
				custom_tooltip = kns_squire.0400.bad_training.tt
				scope:my_squire = {
					kns_add_squire_trait_xp_effect = { MIN_VALUE = -2 MAX_VALUE = 2 }
					send_interface_message = {
						type = event_toast_effect_bad
						title = kns_squire.0400.had_a_training_session.t
